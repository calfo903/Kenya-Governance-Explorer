    52→    case 'BarChart3': return <BarChart3 {...props} />;
    53→    case 'FileText': return <FileText {...props} />;

    56→    case 'Globe': return <Globe {...props} />;

    58→    case 'Users': return <Users {...props} />;
    59→    case 'Database': return <Database {...props} />;
