    71→function SourceIcon({ name, className }: { name: string; className?: string }) {
    72→  const props = { className: className || 'h-5 w-5' };
    73→  switch (name) {
    74→    case 'Scale': return <Scale {...props} />;
    75→    case 'BarChart3': return <BarChart3 {...props} />;
    76→    case 'FileText': return <FileText {...props} />;
    77→    case 'Shield': return <Shield {...props} />;
    78→    case 'Landmark': return <Landmark {...props} />;
    79→    case 'Globe': return <Globe {...props} />;
    80→    case 'Building2': return <Building2 {...props} />;
    81→    case 'Users': return <Users {...props} />;
    82→    case 'Database': return <Database {...props} />;
    83→    case 'CheckCircle2': return <CheckCircle2 {...props} />;
    84→    case 'Search': return <Search {...props} />;
    85→    default: return <Library {...props} />;
    86→  }
    87→}
