   140→function SourceIcon({ name, className }: { name: string; className?: string }) {
   141→  const props = { className: className || 'h-5 w-5' };
   142→  switch (name) {
   143→    case 'Scale': return <Scale {...props} />;
   144→    case 'BarChart3': return <BarChart3 {...props} />;
   145→    case 'FileText': return <FileText {...props} />;
   146→    case 'Shield': return <Shield {...props} />;
   147→    case 'Landmark': return <Landmark {...props} />;
   148→    case 'Globe': return <Globe {...props} />;
   149→    case 'Building2': return <Building2 {...props} />;
   150→    case 'Users': return <Users {...props} />;
   151→    case 'Database': return <Database {...props} />;
   152→    case 'CheckCircle2': return <CheckCircle2 {...props} />;
   153→    case 'Search': return <Search {...props} />;
   154→    default: return <Library {...props} />;
   155→  }
