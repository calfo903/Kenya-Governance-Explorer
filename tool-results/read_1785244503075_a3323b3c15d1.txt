   152→function SourceIcon({ name, className }: { name: string; className?: string }) {
   153→  const props = { className: className || 'h-5 w-5' };
   154→  switch (name) {
   155→    case 'Scale': return <Scale {...props} />;
   156→    case 'BarChart3': return <BarChart3 {...props} />;
   157→    case 'FileText': return <FileText {...props} />;
   158→    case 'Shield': return <Shield {...props} />;
   159→    case 'Landmark': return <Landmark {...props} />;
   160→    case 'Globe': return <Globe {...props} />;
   161→    case 'Building2': return <Building2 {...props} />;
   162→    case 'Users': return <Users {...props} />;
   163→    case 'Database': return <Database {...props} />;
   164→    case 'CheckCircle2': return <CheckCircle2 {...props} />;
   165→    case 'Search': return <Search {...props} />;
   166→    default: return <Library {...props} />;
   167→  }
