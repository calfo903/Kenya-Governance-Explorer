   153→function SourceIcon({ name, className }: { name: string; className?: string }) {
   154→  const props = { className: className || 'h-5 w-5' };
   155→  switch (name) {
   156→    case 'Scale': return <Scale {...props} />;
   157→    case 'BarChart3': return <BarChart3 {...props} />;
   158→    case 'FileText': return <FileText {...props} />;
   159→    case 'Shield': return <Shield {...props} />;
   160→    case 'Landmark': return <Landmark {...props} />;
   161→    case 'Globe': return <Globe {...props} />;
   162→    case 'Building2': return <Building2 {...props} />;
   163→    case 'Users': return <Users {...props} />;
   164→    case 'Database': return <Database {...props} />;
   165→    case 'CheckCircle2': return <CheckCircle2 {...props} />;
   166→    case 'Search': return <Search {...props} />;
   167→    default: return <Library {...props} />;
   168→  }
