   883→/** Count records by opinion type for a given financial year and arm */
   884→export function countOpinions(
   885→  financialYear: string,
   886→  arm: 'executive' | 'assembly',
   887→): Record<string, number> {
   888→  const records = countyAuditData.filter((r) => r.financialYear === financialYear);
   889→  const counts: Record<string, number> = {
   890→    Unmodified: 0,
   891→    Qualified: 0,
   892→    Adverse: 0,
   893→    Disclaimer: 0,
   894→  };
   895→  for (const r of records) {
   896→    const opinion = arm === 'executive' ? r.executiveOpinion : r.assemblyOpinion;
   897→    if (opinion && opinion in counts) {
   898→      counts[opinion]++;
   899→    }
   900→  }
   901→  return counts;
   902→}
