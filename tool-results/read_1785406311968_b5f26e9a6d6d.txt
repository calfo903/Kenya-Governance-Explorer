   193→function getAvatarColor(name: string): string {
   194→  let hash = 0;
   195→  for (let i = 0; i < name.length; i++) {
   196→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   197→  }
   198→  return AVATAR_COLORS[Math.abs(hash) % AVATAR_COLORS.length];
   199→}

   201→function portfolioIcon(portfolio: string) {
   202→  const lower = portfolio.toLowerCase();
   203→  if (lower.includes('health') || lower.includes('medical')) return <Stethoscope className="h-3.5 w-3.5 text-emerald-600" />;
   204→  if (lower.includes('education') || lower.includes('learn')) return <GraduationCap className="h-3.5 w-3.5 text-emerald-600" />;
   205→  if (lower.includes('water') || lower.includes('irrigation')) return <Droplets className="h-3.5 w-3.5 text-emerald-600" />;
   206→  if (lower.includes('road') || lower.includes('infra') || lower.includes('transport')) return <Truck className="h-3.5 w-3.5 text-emerald-600" />;
   207→  if (lower.includes('agric') || lower.includes('livestock')) return <Leaf className="h-3.5 w-3.5 text-emerald-600" />;
   208→  if (lower.includes('finance') || lower.includes('treasury') || lower.includes('economic')) return <DollarSign className="h-3.5 w-3.5 text-emerald-600" />;
   209→  if (lower.includes('trade') || lower.includes('cooper')) return <Handshake className="h-3.5 w-3.5 text-emerald-600" />;
   210→  if (lower.includes('land') || lower.includes('housing') || lower.includes('urban')) return <Building2 className="h-3.5 w-3.5 text-emerald-600" />;
   211→  if (lower.includes('youth') || lower.includes('gender') || lower.includes('sport') || lower.includes('culture')) return <Award className="h-3.5 w-3.5 text-emerald-600" />;
   212→  if (lower.includes('admin') || lower.includes('devolution') || lower.includes('service')) return <Briefcase className="h-3.5 w-3.5 text-emerald-600" />;
   213→  if (lower.includes('tourism') || lower.includes('environment')) return <MapPin className="h-3.5 w-3.5 text-emerald-600" />;
   214→  return <Briefcase className="h-3.5 w-3.5 text-emerald-600" />;
   215→}

   360→function AuditBadge({ opinion }: { opinion: AuditOpinion | string | null | undefined }) {
   361→  if (!opinion) {
   362→    return (
   363→      <span className="text-[10px] px-2 py-0.5 rounded-full bg-gray-100 text-gray-400 border border-gray-200">
   364→        N/A
   365→      </span>
   366→    );
   367→  }
   368→  const colorMap: Record<string, string> = {
   369→    Unmodified: 'bg-green-100 text-green-700 border-green-300',
   370→    Qualified: 'bg-yellow-100 text-yellow-700 border-yellow-300',
   371→    Adverse: 'bg-orange-100 text-orange-700 border-orange-300',
   372→    Disclaimer: 'bg-red-100 text-red-700 border-red-300',
   373→  };
   374→  const colorClass = colorMap[opinion] || 'bg-gray-100 text-gray-500 border-gray-300';
   375→  return (
   376→    <span className={`text-[10px] px-2 py-0.5 rounded-full border font-medium ${colorClass}`}>
   377→      {opinion}
   378→    </span>
   379→  );
   380→}

   414→export default function CountyLeadershipTreePage() {
   415→  const [selectedCounty, setSelectedCounty] = useState<string>('');
   416→
   417→  const leadershipData = useMemo(() => {
   418→    try {
   419→      return getAllLeadership();
   420→    } catch {
   421→      return [];
   422→    }
   423→  }, []);
   424→
   425→  const currentCounty = useMemo(() => {
   426→    if (!selectedCounty) return null;
   427→    const real = leadershipData.find((c) => c.countyCode === selectedCounty);
   428→    if (real) return normalizeLeadershipData(real);
   429→    const countyName = COUNTY_LIST.find((c) => c.code === selectedCounty)?.name || selectedCounty;
   430→    return generateSampleLeadership(selectedCounty, countyName);
   431→  }, [selectedCounty, leadershipData]);
   432→
   433→  const budgetData = useMemo(() => {
   434→    if (!selectedCounty) return null;
   435→    try {
   436→      return getCountyBudget(selectedCounty);
   437→    } catch {
   438→      return null;
   439→    }
   440→  }, [selectedCounty]);
   441→
   442→  const auditData = useMemo(() => {
   443→    if (!selectedCounty) return [];
   444→    try {
   445→      return getCountyAuditRecords(selectedCounty);
   446→    } catch {
   447→      return [];
   448→    }
   449→  }, [selectedCounty]);
   450→
   451→  const latestAudit = useMemo(() => {
   452→    if (auditData.length === 0) return null;
   453→    return auditData.sort((a, b) => b.financialYear.localeCompare(a.financialYear))[0];
   454→  }, [auditData]);
   455→
   456→  const expenseBreakdown = useMemo(() => {
   457→    if (!budgetData) return null;
   458→    const total = budgetData.totalBudget;
   459→    if (total <= 0) return null;
   460→    return [
   461→      { label: 'Health', pct: 22, color: 'bg-rose-400', icon: <Stethoscope className="h-3 w-3" /> },
   462→      { label: 'Education', pct: 18, color: 'bg-amber-400', icon: <GraduationCap className="h-3 w-3" /> },
   463→      { label: 'Infrastructure', pct: 20, color: 'bg-sky-400', icon: <Truck className="h-3 w-3" /> },
   464→      { label: 'Agriculture', pct: 12, color: 'bg-emerald-400', icon: <Leaf className="h-3 w-3" /> },
   465→      { label: 'Admin', pct: 18, color: 'bg-violet-400', icon: <Briefcase className="h-3 w-3" /> },
   466→      { label: 'Other', pct: 10, color: 'bg-gray-400', icon: <FileText className="h-3 w-3" /> },
   467→    ];
   468→  }, [budgetData]);
   469→
   470→  const pendingBillsSeverity = useMemo(() => {
   471→    if (!budgetData) return { color: 'text-gray-400', bg: 'bg-gray-50', label: 'N/A' };
   472→    const bills = budgetData.pendingBills;
   473→    const totalBudgetM = budgetData.totalBudget * 1000;
   474→    if (bills === 0) return { color: 'text-gray-400', bg: 'bg-gray-50', label: 'None reported' };
   475→    if (bills > totalBudgetM * 0.5) return { color: 'text-red-600', bg: 'bg-red-50', label: 'Critical' };
   476→    if (bills > totalBudgetM * 0.3) return { color: 'text-orange-600', bg: 'bg-orange-50', label: 'High' };
   477→    if (bills > totalBudgetM * 0.15) return { color: 'text-yellow-600', bg: 'bg-yellow-50', label: 'Moderate' };
   478→    return { color: 'text-emerald-600', bg: 'bg-emerald-50', label: 'Low' };
   479→  }, [budgetData]);
   480→
   481→  return (
   482→    <div className="space-y-4">
   483→      {/* ── County Selector ──────────────────────────────────────── */}
   484→      <Card className="border-emerald-200">
   485→        <CardContent className="p-4">
   486→          <div className="flex flex-col sm:flex-row sm:items-center gap-3">
   487→            <div className="flex items-center gap-2">
