   193→   193→function getAvatarColor(name: string): string {
   194→   194→  let hash = 0;
   195→   195→  for (let i = 0; i < name.length; i++) {
   196→   196→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   197→   197→  }
   198→   198→  return AVATAR_COLORS[Math.abs(hash) % AVATAR_COLORS.length];
   199→   199→}
