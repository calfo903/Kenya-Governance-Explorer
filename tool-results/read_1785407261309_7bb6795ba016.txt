   154→function SourceIcon({ name, className }: { name: string; className?: string }) {
   155→  const props = { className: className || 'h-5 w-5' };
   156→  switch (name) {
   157→    case 'Scale': return <Scale {...props} />;
   158→    case 'BarChart3': return <BarChart3 {...props} />;
   159→    case 'FileText': return <FileText {...props} />;
   160→    case 'Shield': return <Shield {...props} />;
   161→    case 'Landmark': return <Landmark {...props} />;
   162→    case 'Globe': return <Globe {...props} />;
   163→    case 'Building2': return <Building2 {...props} />;
   164→    case 'Users': return <Users {...props} />;
   165→    case 'Database': return <Database {...props} />;
   166→    case 'CheckCircle2': return <CheckCircle2 {...props} />;
   167→    case 'Search': return <Search {...props} />;
   168→    default: return <Library {...props} />;
   169→  }
