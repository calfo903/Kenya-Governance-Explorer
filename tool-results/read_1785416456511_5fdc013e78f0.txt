   137→// ─── Helper: interpolate color from red to green ────────────────────
   138→
   139→function lerpColor(t: number): string {
   140→  const clamped = Math.max(0, Math.min(1, t));
   141→  const r = Math.round(220 * (1 - clamped) + 22 * clamped);
   142→  const g = Math.round(38 * (1 - clamped) + 163 * clamped);
   143→  const b = Math.round(38 * (1 - clamped) + 74 * clamped);
   144→  return `rgb(${r}, ${g}, ${b})`;
   145→}
   146→
   147→// ─── Helper: population density color ───────────────────────────────
   148→
   149→function densityColor(pop: number, area: number): string {
   150→  const density = pop / area;
   151→  if (density > 2000) return "#7f1d1d";
   152→  if (density > 1000) return "#991b1b";
   153→  if (density > 500) return "#b91c1c";
   154→  if (density > 200) return "#c2410c";
   155→  if (density > 100) return "#ea580c";
   156→  if (density > 50) return "#f59e0b";
   157→  if (density > 20) return "#fbbf24";
   158→  return "#fef08a";
   159→}

   189→      m[r.countyCode] = r.devAbsorptionRate;
   190→    }
   191→  }
   192→  return m;

   312→// ─── County color resolver ──────────────────────────────────────────
   313→
   314→function resolveCountyFill(
   315→  code: string,
   316→  mode: ColorMode,
   317→  govMap: Record<string, (typeof all47Governors)[number]>,
   318→  auditMap: Record<string, (typeof countyAuditData)[number]>,
   319→  budgetMap: Record<string, number>,
   320→): string {
   321→  switch (mode) {
   322→    case "coalition": {
   323→      const gov = govMap[code];
   324→      return gov ? (COALITION_COLORS[gov.coalition] ?? DEFAULT_FILL) : DEFAULT_FILL;
   325→    }
   326→    case "region": {
   327→      const gov = govMap[code];
   328→      return gov ? (REGION_COLORS[gov.region] ?? DEFAULT_FILL) : DEFAULT_FILL;
   329→    }
   330→    case "audit": {
   331→      const audit = auditMap[code];
   332→      if (!audit || !audit.executiveOpinion) return DEFAULT_FILL;
   333→      return AUDIT_COLORS[audit.executiveOpinion] ?? DEFAULT_FILL;
   334→    }
   335→    case "budget": {
   336→      const rate = budgetMap[code];
   337→      if (rate == null) return DEFAULT_FILL;
   338→      return lerpColor(rate / 100);
   339→    }
   340→    case "population": {
   341→      const gov = govMap[code];
   342→      if (!gov) return DEFAULT_FILL;
   343→      return densityColor(gov.population, gov.areaSqKm);
   344→    }
   345→    default:
   346→      return DEFAULT_FILL;
   347→  }
   348→}

   359→  animated = true,
   360→}: KenyaCountyMapProps) {
   361→  const [hoveredCounty, setHoveredCounty] = useState<string | null>(null);
   362→  const [tooltipPos, setTooltipPos] = useState({ x: 0, y: 0 });
   363→  const [tooltipData, setTooltipData] = useState<TooltipData | null>(null);
   364→
   365→  const govMap = useMemo(() => buildGovernorMap(), []);
   366→  const auditMap = useMemo(() => buildAuditMap(), []);
   367→  const budgetMap = useMemo(() => buildBudgetAbsorptionMap(), []);
   368→
   369→  const legendItems = useMemo(() => getLegendItems(colorMode), [colorMode]);
   370→
   371→  const handleMouseEnter = useCallback(
   372→    (county: CountyShape, e: React.MouseEvent<SVGPathElement>) => {
   373→      setHoveredCounty(county.code);
   374→      onCountyHover?.(county.code);
   375→
   376→      const gov = govMap[county.code];
   377→      const audit = auditMap[county.code];
   378→      const budget = getCountyBudget(county.code, "FY 2024/25");
   379→
   380→      const container = (e.currentTarget as SVGPathElement)
   381→        .closest("div")
   382→        ?.getBoundingClientRect();
   383→      if (container) {
   384→        setTooltipPos({
   385→          x: e.clientX - container.left + 15,
   386→          y: e.clientY - container.top - 10,
   387→        });
   388→      }
   389→
   390→      setTooltipData({
   391→        countyName: county.name,
   392→        governor: gov?.name ?? "N/A",
   393→        party: gov?.party ?? "",
   394→        coalition: gov?.coalition ?? "",
   395→        region: gov?.region ?? "",
   396→        population: gov?.population ?? 0,
   397→        auditOpinion: audit?.executiveOpinion ?? null,
   398→        budgetAbsorption: budget?.devAbsorptionRate ?? null,
   399→      });
   400→    },
   401→    [govMap, auditMap, onCountyHover],
   402→  );
   403→
   404→  const handleMouseLeave = useCallback(() => {
   405→    setHoveredCounty(null);
   406→    setTooltipData(null);
   407→    onCountyHover?.(null);
   408→  }, [onCountyHover]);
   409→
   410→  const handleClick = useCallback(
   411→    (county: CountyShape) => {
   412→      onCountyClick?.(county.code, county.name);
   413→    },
   414→    [onCountyClick],
   415→  );
   416→
   417→  return (
   418→    <div className={`relative w-full ${className}`}>
   419→      {/* Tooltip */}
   420→      {tooltipData && hoveredCounty && (
   421→        <MapTooltip data={tooltipData} x={tooltipPos.x} y={tooltipPos.y} />
   422→      )}
   423→
   424→      {/* SVG Map */}
   425→      <svg
   426→        viewBox="0 0 1000 1200"
   427→        className="w-full h-auto"
   428→        role="img"
   429→        aria-label="Map of Kenya showing 47 counties"
   430→      >
   431→        <defs>
   432→          <filter id="county-shadow" x="-5%" y="-5%" width="110%" height="110%">
   433→            <feDropShadow dx="0" dy="1" stdDeviation="1.5" floodOpacity="0.15" />
   434→          </filter>
   435→          <style>{`
   436→            .county-path {
   437→              stroke: #ffffff;
   438→              stroke-width: 1.2;
   439→              cursor: pointer;
   440→              transition: fill 0.2s ease, stroke-width 0.2s ease, filter 0.2s ease, opacity 0.2s ease;
   441→            }
   442→            .county-path:hover {
   443→              stroke-width: 2.5;
   444→              stroke: #0f172a;
   445→              filter: url(#county-shadow);
   446→            }
   447→            .county-path.highlighted {
   448→              stroke-width: 2.5;
   449→              stroke: #0f172a;
   450→            }
   451→            .county-label {
   452→              font-family: ui-sans-serif, system-ui, sans-serif;
   453→              font-size: 9px;
   454→              fill: #0f172a;
   455→              pointer-events: none;
   456→              text-anchor: middle;
   457→              dominant-baseline: central;
   458→              font-weight: 500;
   459→            }
   460→            .county-label-small {
   461→              font-family: ui-sans-serif, system-ui, sans-serif;
   462→              font-size: 7px;
   463→              fill: #334155;
   464→              pointer-events: none;
   465→              text-anchor: middle;
   466→              dominant-baseline: central;
   467→              font-weight: 400;
   468→            }
   469→            @keyframes pulse-county {
   470→              0%, 100% { opacity: 1; }
   471→              50% { opacity: 0.6; }
   472→            }
   473→            .county-animated {
   474→              animation: pulse-county 2s ease-in-out infinite;
   475→            }
   476→          `}</style>
   477→        </defs>
   478→
   479→        {/* Kenya outline border */}
   480→        <path
   481→          d={KENYA_OUTLINE}
   482→          fill="none"
   483→          stroke="#1e293b"
   484→          strokeWidth="2.5"
   485→          strokeLinejoin="round"
   486→        />
   487→
   488→        {/* County shapes */}
   489→        {kenyaCountyPaths.map((county) => {
   490→          const fill = resolveCountyFill(
   491→            county.code,
   492→            colorMode,
   493→            govMap,
   494→            auditMap,
   495→            budgetMap,
   496→          );
   497→          const isHovered = hoveredCounty === county.code;
   498→          const isHighlighted = highlightedCounties.includes(county.code);
   499→          const isSmall = SMALL_COUNTY_CODES.has(county.code);
   500→
   501→          return (
   502→            <g key={county.code}>
   503→              <path
   504→                d={county.path}
   505→                fill={fill}
   506→                className={[
   507→                  "county-path",
   508→                  isHovered ? "" : "",
   509→                  isHighlighted ? "highlighted" : "",
   510→                  animated && isHighlighted ? "county-animated" : "",
   511→                ].join(" ")}
   512→                style={{
   513→                  opacity: isHovered ? 0.85 : 1,
   514→                }}
   515→                onMouseEnter={(e) => handleMouseEnter(county, e)}
   516→                onMouseLeave={handleMouseLeave}
   517→                onClick={() => handleClick(county)}
   518→                aria-label={`${county.name} County`}
   519→              />
   520→              {showLabels && !isSmall && (
   521→                <text
   522→                  x={county.cx}
   523→                  y={county.cy}
   524→                  className={isSmall ? "county-label-small" : "county-label"}
   525→                >
   526→                  {county.name.length > 10
   527→                    ? county.name.split(" ").map((w, i) => (
   528→                        <tspan
   529→                          key={i}
   530→                          x={county.cx}
   531→                          dy={i === 0 ? "-0.4em" : "1.1em"}
   532→                        >
   533→                          {w}
   534→                        </tspan>
   535→                      ))
   536→                    : county.name}
   537→                </text>
   538→              )}
   539→            </g>
   540→          );
   541→        })}
   542→      </svg>
   543→
   544→      {/* Legend */}
   545→      <div className="mt-3 flex flex-wrap items-center justify-center gap-x-4 gap-y-2 px-2">
   546→        <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   547→          <Info className="h-3 w-3" />
   548→          <span className="font-medium capitalize">
   549→            {colorMode === "coalition"
   550→              ? "Coalition"
   551→              : colorMode === "region"
   552→                ? "Region"
   553→                : colorMode === "audit"
   554→                  ? "Audit Opinion"
   555→                  : colorMode === "budget"
   556→                    ? "Budget Absorption"
   557→                    : "Population Density"}
   558→          </span>
   559→        </div>
   560→        {legendItems.map((item, i) => (
   561→          <div key={i} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   562→            <span
   563→              className="inline-block h-3 w-3 rounded-sm border border-white/30 shadow-sm"
   564→              style={{ backgroundColor: item.color }}
   565→            />
   566→            <span>{item.label}</span>
   567→          </div>
   568→        ))}
   569→      </div>
   570→    </div>
   571→  );
   572→}
