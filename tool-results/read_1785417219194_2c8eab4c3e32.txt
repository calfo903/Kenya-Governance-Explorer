   147→  const [multiSelectedCounties, setMultiSelectedCounties] = useState<string[]>([]);
   148→  const [quickFilter, setQuickFilter] = useState<string | null>(null);
   149→
   150→  const FINANCIAL_YEARS = ['FY 2022/23', 'FY 2023/24', 'FY 2024/25'];
   151→
   152→  const governorMap = useMemo(() => {
   153→    const map = new Map<string, GovernorEntry>();
   154→    all47Governors.forEach(g => map.set(g.county, g));
   155→    return map;
   156→  }, []);
   157→
   158→  const getColor = (shape: CountyShape) => {
   159→    const gov = governorMap.get(shape.name);
   160→    if (colorMode === 'coalition') return gov ? COALITION_COLORS[gov.coalition] || '#9ca3af' : '#9ca3af';
   161→    if (colorMode === 'region') return gov ? REGION_COLORS[gov.region] || '#9ca3af' : '#9ca3af';
   162→    if (colorMode === 'audit') {
   163→      const auditKey = SIMULATED_AUDIT[shape.name] || 'qualified';
   164→      return AUDIT_COLORS[auditKey];
   165→    }
   166→    return '#9ca3af';
   167→  };
   168→
   169→  const filteredShapes = useMemo(() => {
   170→    let shapes = countyShapes;
   171→    if (searchQuery) {
   172→      const q = searchQuery.toLowerCase();
   173→      shapes = shapes.filter(s => s.name.toLowerCase().includes(q));
   174→    }
   175→    if (regionFilter !== 'all') {
   176→      shapes = shapes.filter(s => governorMap.get(s.name)?.region === regionFilter);
   177→    }
   178→    if (coalitionFilter !== 'all' && colorMode === 'coalition') {
   179→      shapes = shapes.filter(s => governorMap.get(s.name)?.coalition === coalitionFilter);
   180→    }
   181→    if (auditFilter !== 'all' && colorMode === 'audit') {
   182→      shapes = shapes.filter(s => SIMULATED_AUDIT[s.name] === auditFilter);
   183→    }
   184→    // Apply quick filter
   185→    if (quickFilter === 'adverse') {
   186→      shapes = shapes.filter(s => {
   187→        const rec = countyAuditData.find(a => a.countyCode === s.code && a.financialYear === selectedFY);
   188→        return rec && rec.executiveOpinion === 'Adverse';
   189→      });
   190→    } else if (quickFilter === 'top5-budget') {
   191→      const budgetForFY = countyBudgetData
   192→        .filter(b => b.financialYear === selectedFY)
   193→        .sort((a, b) => b.devAbsorptionRate - a.devAbsorptionRate)
   194→        .slice(0, 5)
   195→        .map(b => b.countyCode);
   196→      shapes = shapes.filter(s => budgetForFY.includes(s.code));
   197→    } else if (quickFilter === 'largest-pop') {
   198→      const top10 = [...all47Governors].sort((a, b) => b.population - a.population).slice(0, 10).map(g => g.code);
   199→      shapes = shapes.filter(s => top10.includes(s.code));
   200→    }
   201→    return shapes;
   202→  }, [regionFilter, coalitionFilter, auditFilter, searchQuery, colorMode, governorMap, quickFilter, selectedFY]);
   203→
   204→  // Quick filter counts
   205→  const quickFilterCounts = useMemo(() => {
   206→    const adverseCount = countyAuditData.filter(a => a.financialYear === selectedFY && a.executiveOpinion === 'Adverse').length;
   207→    return {
   208→      'adverse': adverseCount,
   209→      'top5-budget': 5,
   210→      'largest-pop': 10,
   211→    };
   212→  }, [selectedFY]);

   237→  const removeMultiSelected = useCallback((code: string) => {
   238→    setMultiSelectedCounties(prev => prev.filter(c => c !== code));

   242→    <div className="space-y-5">
   243→      {/* Hero */}
   244→      <div className="bg-gradient-to-br from-slate-800 to-slate-900 rounded-xl p-6 text-white">
   245→        <div className="flex items-start gap-4">
