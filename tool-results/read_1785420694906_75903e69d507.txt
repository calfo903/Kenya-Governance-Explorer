   169→  const props = { className: className || 'h-5 w-5' };

   171→    case 'Scale': return <Scale {...props} />;

   173→    case 'FileText': return <FileText {...props} />;

   175→    case 'Landmark': return <Landmark {...props} />;
   176→    case 'Globe': return <Globe {...props} />;
   177→    case 'Building2': return <Building2 {...props} />;
   178→    case 'Users': return <Users {...props} />;
   179→    case 'Database': return <Database {...props} />;
   180→    case 'CheckCircle2': return <CheckCircle2 {...props} />;
   181→    case 'Search': return <Search {...props} />;
