   258→// ─── Profile Card Component ────────────────────────────────────────────
   259→
   260→interface RepresentativeProfile {
   261→  name: string;
   262→  party?: string;
   263→  coalition?: string;
   264→  qualification?: string;
   265→}
   266→
   267→interface RepresentativePreviewCardProps {
   268→  roleKey: string;
   269→  profile?: RepresentativeProfile | null;
   270→  wardName?: string;       // For MCA cards — show which ward
   271→  constituencyName?: string; // For MP/MCA — show which constituency
   272→  compact?: boolean;       // If true, show minimal view
   273→}
   274→
   275→function getInitials(name: string): string {
   276→  return name
   277→    .split(' ')
   278→    .filter(Boolean)
   279→    .map((w) => w[0])
   280→    .slice(0, 2)
   281→    .join('')
   282→    .toUpperCase();
   283→}
   284→
   285→const AVATAR_COLORS: Record<string, string> = {
   286→  governor: 'bg-amber-100 text-amber-700 ring-amber-300',
   287→  senator: 'bg-indigo-100 text-indigo-700 ring-indigo-300',
   288→  womanRep: 'bg-rose-100 text-rose-700 ring-rose-300',
   289→  mp: 'bg-emerald-100 text-emerald-700 ring-emerald-300',
   290→  mca: 'bg-teal-100 text-teal-700 ring-teal-300',
   291→};
   292→
   293→export function RepresentativePreviewCard({
   294→  roleKey,
   295→  profile,
   296→  wardName,
   297→  constituencyName,
   298→  compact = false,
   299→}: RepresentativePreviewCardProps) {
   300→  const mandate = ROLE_MANDATES[roleKey];
   301→  if (!mandate) return null;
   302→
   303→  const isPlaceholder = !profile || profile.name.includes('(TBD)') || profile.name.includes('TBD');
   304→  const avatarColor = AVATAR_COLORS[roleKey] || 'bg-gray-100 text-gray-700 ring-gray-300';
   305→
   306→  if (compact) {
   307→    // Compact card: just profile + key duties
   308→    return (
   309→      <Card className={`border ${mandate.borderClass} overflow-hidden`}>
   310→        <CardContent className="p-3">
   311→          <div className="flex items-start gap-3">
   312→            {/* Avatar */}
   313→            <div
   314→              className={`h-10 w-10 rounded-full flex items-center justify-center text-xs font-bold ring-2 shrink-0 ${
   315→                isPlaceholder ? 'bg-gray-100 text-gray-400 ring-gray-200' : avatarColor
   316→              }`}
   317→            >
   318→              {isPlaceholder ? '?' : getInitials(profile.name)}
   319→            </div>
   320→            <div className="min-w-0 flex-1">
   321→              <div className="flex items-center gap-1.5 flex-wrap">
   322→                <span className={`text-xs font-bold ${mandate.colorClass}`}>
   323→                  {mandate.roleTitle}
   324→                </span>
   325→                {wardName && (
   326→                  <Badge variant="outline" className="text-[8px] px-1 py-0 h-3.5 text-teal-600 border-teal-200">
   327→                    {wardName}
   328→                  </Badge>
   329→                )}
   330→                {constituencyName && (
   331→                  <Badge variant="outline" className="text-[8px] px-1 py-0 h-3.5 text-emerald-600 border-emerald-200">
   332→                    {constituencyName}
   333→                  </Badge>
   334→                )}
   335→              </div>
   336→              <p className={`text-xs font-medium truncate mt-0.5 ${isPlaceholder ? 'text-gray-400 italic' : 'text-gray-800'}`}>
   337→                {profile?.name || `${mandate.roleTitle} — TBD`}
   338→              </p>
   339→              <div className="flex items-center gap-2 mt-0.5">
   340→                {profile?.party && !isPlaceholder && (
   341→                  <span className="text-[10px] text-gray-500">{profile.party}</span>
   342→                )}
   343→                {profile?.coalition && !isPlaceholder && (
   344→                  <Badge variant="outline" className="text-[8px] px-1 py-0 h-3.5 border-gray-200 text-gray-500">

   346→                  </Badge>
   347→                )}
   348→              </div>
   349→            </div>
   350→          </div>
   351→          {/* Key duties (compact) */}
   352→          <div className="mt-2.5 space-y-1">
   353→            {mandate.duties.slice(0, 3).map((duty, i) => (
   354→              <div key={i} className="flex items-start gap-1.5">
   355→                {duty.icon}
   356→                <span className="text-[10px] text-gray-600 leading-tight">{duty.title}</span>
   357→              </div>
   358→            ))}
   359→            {mandate.duties.length > 3 && (
   360→              <span className="text-[9px] text-gray-400 italic">+{mandate.duties.length - 3} more duties</span>
   361→            )}
   362→          </div>
   363→        </CardContent>
   364→      </Card>
   365→    );
   366→  }
   367→
   368→  // Full card: profile + all duties + constitutional basis

   370→    <Card className={`border ${mandate.borderClass} overflow-hidden`}>
   371→      {/* Role Header */}
   372→      <div className={`${mandate.bgClass} px-4 py-3 border-b ${mandate.borderClass}`}>
   373→        <div className="flex items-center justify-between">
   374→          <div className="flex items-center gap-2">
   375→            <div className={`${mandate.colorClass}`}>{mandate.icon}</div>
   376→            <div>
   377→              <h4 className={`text-sm font-bold ${mandate.colorClass}`}>{mandate.roleTitle}</h4>
   378→              <p className="text-[10px] text-gray-500">{mandate.constitutionalBasis}</p>
   379→            </div>
   380→          </div>
   381→          <div className="text-right">
   382→            <p className="text-[10px] text-gray-500">Term: {mandate.termLength}</p>
   383→            <p className="text-[10px] text-gray-500">{mandate.salaryRange}</p>
   384→          </div>
   385→        </div>
   386→      </div>
   387→
   388→      <CardContent className="p-4 space-y-3">
   389→        {/* Profile Section */}
   390→        <div className="flex items-start gap-3">
   391→          {/* Avatar */}
   392→          <div

   394→              isPlaceholder ? 'bg-gray-100 text-gray-400 ring-gray-200' : avatarColor
   395→            }`}
   396→          >
   397→            {isPlaceholder ? '?' : getInitials(profile?.name || '')}
   398→          </div>
   399→          <div className="min-w-0 flex-1">
   400→            <p className={`text-sm font-bold ${isPlaceholder ? 'text-gray-400 italic' : 'text-gray-900'}`}>
   401→              {profile?.name || `${mandate.roleTitle} — To Be Determined`}
   402→            </p>
   403→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   404→              {profile?.party && !isPlaceholder && (
   405→                <Badge variant="outline" className="text-[9px] h-5 border-gray-200">
   406→                  {profile.party}

   409→              {profile?.coalition && !isPlaceholder && (

   541→        />
