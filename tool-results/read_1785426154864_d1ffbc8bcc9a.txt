   172→  const props = { className: className || 'h-5 w-5' };
   173→  switch (name) {
   174→    case 'Scale': return <Scale {...props} />;

   176→    case 'FileText': return <FileText {...props} />;

   179→    case 'Globe': return <Globe {...props} />;
   180→    case 'Building2': return <Building2 {...props} />;
   181→    case 'Users': return <Users {...props} />;
   182→    case 'Database': return <Database {...props} />;

   184→    case 'Search': return <Search {...props} />;
