   156→
   157→function getInitials(name: string): string {
   158→  const cleaned = name.replace(/^(Hon\.?|Dr\.?|Maj\.?|Eng\.?|Prof\.?)\s*/i, '');
   159→  const parts = cleaned.split(/\s+/).filter(Boolean);
   160→  if (parts.length >= 2) return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   161→  return parts[0] ? parts[0].substring(0, 2).toUpperCase() : '??';
   162→}
   163→
   164→function formatTerm(start: string, end: string): string {
   165→  const fmt = (d: string) => {
   166→    const date = new Date(d);
   167→    return date.toLocaleDateString('en-KE', { month: 'short', year: 'numeric' });
   168→  };
   169→  return `${fmt(start)} – ${fmt(end)}`;
   170→}
   171→
   172→function shortenCoalition(coalition: string): string {
   173→  if (coalition.includes('Kenya Kwanza')) return 'Kenya Kwanza';
   174→  if (coalition.includes('Azimio')) return 'Azimio';
   175→  if (coalition === 'Independent') return 'Independent';
   176→  return coalition.length > 30 ? coalition.substring(0, 28) + '…' : coalition;
   177→}

   179→// ─── BIO GENERATION ──────────────────────────────────────────────

   181→function getGovernorBio(name: string, county: string, party: string): string {
   182→  return `Elected Governor of ${county} County in the August 2022 General Election on a ${party} ticket. As county chief executive, responsible for implementing county legislation, coordinating county administration, and delivering services to the people of ${county}.`;
   183→}
   184→
   185→function getSenatorBio(name: string, county: string, party: string): string {
   186→  return `Elected Senator for ${county} County in August 2022. Represents the county at the Senate, protecting county interests, overseeing revenue allocation, and participating in the legislative process at the national level.`;
   187→}
   188→
   189→function getWomanRepBio(name: string, county: string, party: string): string {
   190→  return `Elected Women Representative for ${county} County in August 2022. Represents women\'s interests at the National Assembly and champions gender-responsive legislation and women\'s participation in governance.`;
   191→}
   192→
   193→function getMpbio(name: string, constituency: string, county: string, party: string): string {
   194→  return `Elected Member of the National Assembly for ${constituency} Constituency, ${county} County, in August 2022. Represents constituency interests in Parliament, debates legislation, and oversees executive action.`;
   195→}

   197→function getMcabio(name: string, ward: string, constituency: string, county: string): string {
