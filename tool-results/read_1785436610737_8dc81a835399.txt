   152→  // Track shift key for multi-select
   153→  const shiftHeldRef = useRef(false);

   155→    const onKeyDown = (e: KeyboardEvent) => { if (e.key === 'Shift') shiftHeldRef.current = true; };

   158→    window.addEventListener('keyup', onKeyUp);
   159→    return () => {
   160→      window.removeEventListener('keydown', onKeyDown);
   161→      window.removeEventListener('keyup', onKeyUp);

   165→  const governorMap = useMemo(() => {
   166→    const map = new Map<string, GovernorEntry>();

   169→  }, []);
   170→
   171→  const getColor = (shape: CountyShape) => {
   172→    const gov = governorMap.get(shape.name);
   173→    if (colorMode === 'coalition') return gov ? COALITION_COLORS[gov.coalition] || '#9ca3af' : '#9ca3af';
   174→    if (colorMode === 'region') return gov ? REGION_COLORS[gov.region] || '#9ca3af' : '#9ca3af';
   175→    if (colorMode === 'audit') {
   176→      const auditKey = SIMULATED_AUDIT[shape.name] || 'qualified';
   177→      return AUDIT_COLORS[auditKey];
   178→    }
   179→    return '#9ca3af';
   180→  };
   181→
   182→  const filteredShapes = useMemo(() => {
   183→    let shapes = countyShapes;
   184→    if (searchQuery) {
   185→      const q = searchQuery.toLowerCase();
   186→      shapes = shapes.filter(s => s.name.toLowerCase().includes(q));
   187→    }
   188→    if (regionFilter !== 'all') {
   189→      shapes = shapes.filter(s => governorMap.get(s.name)?.region === regionFilter);
   190→    }
   191→    if (coalitionFilter !== 'all' && colorMode === 'coalition') {
   192→      shapes = shapes.filter(s => governorMap.get(s.name)?.coalition === coalitionFilter);
   193→    }
   194→    if (auditFilter !== 'all' && colorMode === 'audit') {
   195→      shapes = shapes.filter(s => SIMULATED_AUDIT[s.name] === auditFilter);
   196→    }
   197→    // Apply quick filter
   198→    if (quickFilter === 'adverse') {
   199→      shapes = shapes.filter(s => {
   200→        const rec = countyAuditData.find(a => a.countyCode === s.code && a.financialYear === selectedFY);
   201→        return rec && rec.executiveOpinion === 'Adverse';
   202→      });
   203→    } else if (quickFilter === 'top5-budget') {
   204→      const budgetForFY = countyBudgetData
   205→        .filter(b => b.financialYear === selectedFY)
   206→        .sort((a, b) => b.devAbsorptionRate - a.devAbsorptionRate)
   207→        .slice(0, 5)
   208→        .map(b => b.countyCode);
   209→      shapes = shapes.filter(s => budgetForFY.includes(s.code));
   210→    } else if (quickFilter === 'largest-pop') {
   211→      const top10 = [...all47Governors].sort((a, b) => b.population - a.population).slice(0, 10).map(g => g.code);
   212→      shapes = shapes.filter(s => top10.includes(s.code));
   213→    }
   214→    return shapes;
