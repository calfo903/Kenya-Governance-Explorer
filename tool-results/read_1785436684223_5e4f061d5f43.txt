   144→// ─── Helper: interpolate color from red to green ────────────────────
   145→
   146→function lerpColor(t: number): string {
   147→  const clamped = Math.max(0, Math.min(1, t));
   148→  const r = Math.round(220 * (1 - clamped) + 22 * clamped);
   149→  const g = Math.round(38 * (1 - clamped) + 163 * clamped);
   150→  const b = Math.round(38 * (1 - clamped) + 74 * clamped);
   151→  return `rgb(${r}, ${g}, ${b})`;
   152→}

   154→// ─── Helper: population density color ───────────────────────────────
   155→
   156→function densityColor(pop: number, area: number): string {
   157→  const density = pop / area;
   158→  if (density > 2000) return "#7f1d1d";
   159→  if (density > 1000) return "#991b1b";
   160→  if (density > 500) return "#b91c1c";
   161→  if (density > 200) return "#c2410c";
   162→  if (density > 100) return "#ea580c";
   163→  if (density > 50) return "#f59e0b";
   164→  if (density > 20) return "#fbbf24";
   165→  return "#fef08a";
   166→}

   176→function buildAuditMap(fy?: string) {
   177→  const m: Record<string, (typeof countyAuditData)[number]> = {};
   178→  for (const r of countyAuditData) {
   179→    if (fy) {
   180→      // Filter to specific financial year
   181→      if (r.financialYear === fy) {
   182→        m[r.countyCode] = r;
   183→      }
   184→    } else {
   185→      // Pick latest
   186→      if (!m[r.countyCode] || r.financialYear > m[r.countyCode].financialYear) {
   187→        m[r.countyCode] = r;
   188→      }
   189→    }
   190→  }
   191→  return m;
   192→}
   193→
   194→function buildBudgetAbsorptionMap(fy?: string) {
   195→  const m: Record<string, number> = {};
   196→  if (fy) {
   197→    // Filter to specific financial year, prefer full year records
   198→    for (const r of countyBudgetData) {
   199→      if (r.financialYear === fy && r.period === "Full Year") {
   200→        m[r.countyCode] = r.devAbsorptionRate;
   201→      }
   202→    }
   203→    // Fall back to any record in that FY if no full year record
   204→    for (const r of countyBudgetData) {
   205→      if (r.financialYear === fy && !m[r.countyCode]) {
   206→        m[r.countyCode] = r.devAbsorptionRate;
   207→      }
   208→    }
   209→  } else {
   210→    // Default: pick FY 2024/25 full year records
   211→    for (const r of countyBudgetData) {
   212→      if (!m[r.countyCode] || r.financialYear > "FY 2024/24") {
   213→        m[r.countyCode] = r.devAbsorptionRate;
   214→      }
   215→    }
   216→    // Ensure we pick the FY 2024/25 full year records
   217→    for (const r of countyBudgetData) {
   218→      if (r.financialYear === "FY 2024/25" && r.period === "Full Year") {
   219→        m[r.countyCode] = r.devAbsorptionRate;
   220→      }
   221→    }
   222→  }
   223→  return m;
   224→}

   301→function getAuditBadgeColor(opinion: string | null): string {

   306→    case AUDIT_OPINIONS.DISCLAIMER: return "bg-red-100 text-red-800 border-red-300";
   307→    default: return "bg-gray-100 text-gray-500 border-gray-300";
   308→  }

   311→function getBudgetBarColor(rate: number | null): string {
   312→  if (rate == null) return "bg-gray-300";

   314→  if (rate >= 50) return "bg-yellow-500";
   315→  if (rate >= 30) return "bg-orange-500";
   316→  return "bg-red-500";
   317→}

   363→              <span>Absorption: {data.budgetAbsorption != null ? `${data.budgetAbsorption}%` : "N/A"}</span>

   375→            </div>

   379→    </div>

   383→// ─── County color resolver ──────────────────────────────────────────
   384→
   385→function resolveCountyFill(
   386→  code: string,
   387→  mode: ColorMode,
   388→  govMap: Record<string, (typeof all47Governors)[number]>,
   389→  auditMap: Record<string, (typeof countyAuditData)[number]>,
   390→  budgetMap: Record<string, number>,
   391→): string {
   392→  switch (mode) {
   393→    case "coalition": {
   394→      const gov = govMap[code];
   395→      return gov ? (COALITION_COLORS[gov.coalition] ?? DEFAULT_FILL) : DEFAULT_FILL;
   396→    }
   397→    case "region": {
   398→      const gov = govMap[code];
   399→      return gov ? (REGION_COLORS[gov.region] ?? DEFAULT_FILL) : DEFAULT_FILL;
   400→    }
   401→    case "audit": {
   402→      const audit = auditMap[code];
   403→      if (!audit || !audit.executiveOpinion) return DEFAULT_FILL;
   404→      return AUDIT_COLORS[audit.executiveOpinion] ?? DEFAULT_FILL;
   405→    }
   406→    case "budget": {
   407→      const rate = budgetMap[code];
   408→      if (rate == null) return DEFAULT_FILL;
   409→      return lerpColor(rate / 100);
   410→    }
   411→    case "population": {
   412→      const gov = govMap[code];
   413→      if (!gov) return DEFAULT_FILL;
   414→      return densityColor(gov.population, gov.areaSqKm);
   415→    }
   416→    default:
   417→      return DEFAULT_FILL;
   418→  }
   419→}
