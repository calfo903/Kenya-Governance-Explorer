   165→function PageContent() {
   166→  const t = useTranslations();
   167→  const locale = useLocaleStore((s) => s.locale);
   168→  const [activeTab, setActiveTab] = useState<TabId>('summary');
   169→  const [selectedCounty, setSelectedCounty] = useState<string>('034');
   170→  const [expandedCounties, setExpandedCounties] = useState<Set<string>>(new Set());
   171→  const [comparisonList, setComparisonList] = useState<ComparisonItem[]>([]);
   172→  const [filters, setFilters] = useState<FilterState>({});
   173→  const [sidebarOpen, setSidebarOpen] = useState(false);
   174→  const commandPaletteRef = React.useRef<{ open: () => void; close: () => void }>(null);
   175→
   176→  // Localized nav items
   177→  const navItems = useMemo(() => navItemDefs.map(def => ({
   178→    ...def,
   179→    label: t(def.labelKey),
   180→  })), [t]);
   181→
   182→  const allCounties = useMemo((): County[] => {
   183→    const placeholders = getPlaceholderCounties();
   184→    return placeholders.map((c) => {
   185→      if (c.code === '034') return kajiadoCounty;
   186→      // Inject real audit opinion from county-audit-data
   187→      const auditRecords = getCountyAuditRecords(c.code);
   188→      const latestAudit = auditRecords.find(a => a.financialYear === 'FY 2024/25');
   189→      const latestBudget = getCountyBudget(c.code, 'FY 2024/25');
   190→      return {
   191→        ...c,
   192→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   193→        executiveAuditSource: latestAudit?.source,
   194→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
   195→        developmentAbsorptionSource: latestBudget ? {
   196→          source: latestBudget.source.source,
   197→          reportTitle: latestBudget.source.reportTitle,
   198→          financialYear: latestBudget.source.financialYear,
   199→          url: latestBudget.source.url,
   200→          accessedDate: latestBudget.source.accessedDate,
   201→        } : undefined,
   202→        dataAvailability: 'partial' as const,
   203→        dataAvailabilityNote: `Audit opinion: ${latestAudit?.executiveOpinion || 'pending'} (FY 2024/25). Budget absorption: ${latestBudget?.devAbsorptionRate ?? 'pending'}%. Full official data from oagkenya.go.ke and cob.go.ke.`,
   204→      };
   205→    });
   206→  }, []);
   207→
   208→  const toggleCounty = (code: string) => {
   209→    const next = new Set(expandedCounties);
   210→    if (next.has(code)) next.delete(code); else next.add(code);
   211→    setExpandedCounties(next);
   212→  };
   213→

   215→    if (comparisonList.length >= 4) return;
   216→    if (comparisonList.find(c => c.representative.id === rep.id)) return;
   217→    setComparisonList([...comparisonList, { representative: rep, countyName }]);
   218→  };
   219→
   220→  const removeFromComparison = (id: string) => {
   221→    setComparisonList(comparisonList.filter(c => c.representative.id !== id));
   222→  };
   223→
   224→  const filteredGovernors = useMemo(() => {
   225→    return all47Governors.filter((g) => {
   226→      if (filters.region && g.region !== filters.region) return false;
   227→      if (filters.coalition && g.coalition !== filters.coalition) return false;
   228→      if (filters.party && g.party !== filters.party) return false;
   229→      if (filters.keyword) {
   230→        const kw = filters.keyword.toLowerCase();
   231→        if (!g.name.toLowerCase().includes(kw) && !g.county.toLowerCase().includes(kw)) return false;
   232→      }
   233→      return true;
   234→    });
   235→  }, [filters]);
   236→
   237→  // Group nav items by section
   238→  const governanceItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.governance');
   239→  const civicItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.civicTools');
   240→  const citizenItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.citizenAction');
   241→  const dataItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.dataAlerts');
   242→  const analyticsItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.analytics');
   243→  const leadershipItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.leadershipProjects');
   244→  const insightItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.insights');
   245→
   246→  // Section labels lookup
   247→  const sectionLabels = useMemo(() => ({
   248→    'nav.sections.governance': t('nav.sections.governance'),
   249→    'nav.sections.civicTools': t('nav.sections.civicTools'),
   250→    'nav.sections.citizenAction': t('nav.sections.citizenAction'),
   251→    'nav.sections.dataAlerts': t('nav.sections.dataAlerts'),
   252→    'nav.sections.analytics': t('nav.sections.analytics'),
   253→    'nav.sections.leadershipProjects': t('nav.sections.leadershipProjects'),
   254→    'nav.sections.insights': t('nav.sections.insights'),
   255→  }), [t]);
   256→
   257→  // Helper to get localized label for a nav item
   258→  const getItemLabel = (id: TabId) => navItems.find(n => n.id === id)?.label ?? id;
   259→
   260→  // Helper to get localized section label
   261→  const getSectionLabel = (sectionKey: string) => sectionLabels[sectionKey] ?? sectionKey;
   262→
   263→  // Sidebar section renderer helper
   264→  const renderSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (

   274→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   275→          }`}
   276→        >
   277→          <item.icon className="h-3.5 w-3.5 shrink-0" />
   278→          <span>{getItemLabel(item.id)}</span>
   279→        </button>
   280→      ))}
   281→    </div>
   282→  );
   283→
   284→  // Mobile sidebar section renderer helper

   286→    <div>
   287→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   288→      {items.map((item) => (
   289→        <button
   290→          key={item.id}
   291→          onClick={() => { setActiveTab(item.id); setSidebarOpen(false); }}
   292→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors mb-0.5 ${
   293→            activeTab === item.id
   294→              ? `${activeColor} text-white`
   295→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   296→          }`}
   297→        >
   298→          <item.icon className="h-4 w-4" />
   299→          <span>{getItemLabel(item.id)}</span>
   300→        </button>
   301→      ))}
   302→    </div>
   303→  );
   304→
   305→  const { theme, setTheme } = useTheme();
   306→
   307→  // Keyboard shortcuts
   308→  React.useEffect(() => {
   309→    const handler = (e: KeyboardEvent) => {
   310→      if (e.key === 'Escape') { setSidebarOpen(false); return; }
   311→      if ((e.ctrlKey || e.metaKey) && e.key === 'k') { e.preventDefault(); commandPaletteRef.current?.open(); return; }
   312→      if ((e.ctrlKey || e.metaKey) && e.key === 'd') { e.preventDefault(); setTheme(theme === 'dark' ? 'light' : 'dark'); return; }
   313→      if (e.key === '?' && !['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) {
   314→        alert('Keyboard Shortcuts:\nCtrl+K: Search\nCtrl+D: Toggle Dark Mode\nEsc: Close panels\n?: This help');
   315→      }
   316→    };
   317→    window.addEventListener('keydown', handler);
   318→    return () => window.removeEventListener('keydown', handler);
   319→  }, [theme, setTheme]);
   320→
   321→  return (
   322→    <TooltipProvider>
   323→      <div className="min-h-screen flex flex-col bg-stone-50 dark:bg-stone-950">
   324→        {/* ══════════ HEADER ══════════ */}
   325→        <header className="bg-white dark:bg-stone-900 border-b border-stone-200 dark:border-stone-700 sticky top-0 z-50">
   326→          <div className="max-w-[1400px] mx-auto px-4 sm:px-6 lg:px-8">
   327→            <div className="py-3 flex items-center justify-between gap-4">
   328→              <div className="flex items-center gap-3">
   329→                <button
   330→                  onClick={() => setSidebarOpen(!sidebarOpen)}
   331→                  className="lg:hidden h-9 w-9 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 transition-colors"
   332→                >
   333→                  <Menu className="h-4 w-4 text-stone-600 dark:text-stone-300" />
   334→                </button>
   335→                <div className="h-8 w-8 rounded-lg bg-emerald-600 flex items-center justify-center shrink-0">
   336→                  <Shield className="h-4 w-4 text-white" />
   337→                </div>
   338→                <div>
   339→                  <h1 className="text-sm sm:text-base font-bold text-stone-900 dark:text-stone-100 tracking-tight leading-tight">
   340→                    {t('app.title')}
   341→                  </h1>
   342→                  <p className="text-[10px] sm:text-xs text-stone-500">
   343→                    {t('app.subtitle')}
   344→                  </p>
   345→                </div>
   346→              </div>
   347→              <div className="flex items-center gap-2">
   348→                {comparisonList.length > 0 && (
   349→                  <Badge variant="secondary" className="text-[10px] h-6 bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300">
   350→                    {t('app.headerCompare', { count: comparisonList.length })}
   351→                  </Badge>
   352→                )}
   353→                <button
   354→                  onClick={() => commandPaletteRef.current?.open()}
   355→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   356→                  title={t('app.searchShortcut')}
   357→                >
   358→                  <Keyboard className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" />
   359→                </button>
   360→                <button
   361→                  onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   362→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   363→                  title={t('app.darkModeShortcut')}
   364→                >
   365→                  {theme === 'dark' ? <Sun className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" /> : <Moon className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" />}
   366→                </button>
   367→                <LanguageToggle />
   368→                <a href="https://kenyalaw.org/" target="_blank" rel="noopener noreferrer" className="hidden sm:flex items-center gap-1 text-[10px] text-stone-400 hover:text-emerald-600 transition-colors">
   369→                  <BookMarked className="h-3 w-3" />{t('common.constitution')}
   370→                </a>
   371→              </div>
   372→            </div>
   373→          </div>
   374→        </header>
   375→
   376→        {/* ══════════ LAYOUT: SIDEBAR + MAIN ══════════ */}
   377→        <div className="flex flex-1">
