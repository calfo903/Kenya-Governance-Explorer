    82→function getOpinionTrendIcon(trend: string) {
    83→  if (trend === 'improved') return <ArrowUpRight className="h-3.5 w-3.5 text-green-600" />;
    84→  if (trend === 'declined') return <ArrowDownRight className="h-3.5 w-3.5 text-red-600" />;
    85→  if (trend === 'same') return <Minus className="h-3.5 w-3.5 text-amber-600" />;
    86→  return <Minus className="h-3.5 w-3.5 text-stone-400" />;
    87→}
    88→
    89→function getOpinionBadgeClass(opinion: string | null) {
    90→  if (!opinion) return 'bg-gray-100 dark:bg-gray-800 text-gray-500 dark:text-gray-400 border-gray-300 dark:border-gray-600';
    91→  switch (opinion) {
    92→    case 'Unmodified': return 'bg-green-100 text-green-800 border-green-300';
    93→    case 'Qualified': return 'bg-yellow-100 text-yellow-800 border-yellow-300';
    94→    case 'Adverse': return 'bg-orange-100 text-orange-800 border-orange-300';
    95→    case 'Disclaimer': return 'bg-red-100 text-red-800 border-red-300';
    96→    default: return 'bg-gray-100 dark:bg-gray-800 text-gray-500 dark:text-gray-400 border-gray-300 dark:border-gray-600';
    97→  }
    98→}
