   165→function PageContent() {
   166→  const t = useTranslations();
   167→  const locale = useLocaleStore((s) => s.locale);
   168→  const [activeTab, setActiveTab] = useState<TabId>('summary');
   169→  const [selectedCounty, setSelectedCounty] = useState<string>('034');
   170→  const [expandedCounties, setExpandedCounties] = useState<Set<string>>(new Set());
   171→  const [comparisonList, setComparisonList] = useState<ComparisonItem[]>([]);
   172→  const [filters, setFilters] = useState<FilterState>({});
   173→  const [sidebarOpen, setSidebarOpen] = useState(false);
   174→  const commandPaletteRef = React.useRef<{ open: () => void; close: () => void }>(null);
   175→
   176→  // Localized nav items
   177→  const navItems = useMemo(() => navItemDefs.map(def => ({
   178→    ...def,
   179→    label: t(def.labelKey),
   180→  })), [t]);
   181→
   182→  const allCounties = useMemo((): County[] => {
   183→    const placeholders = getPlaceholderCounties();
   184→    return placeholders.map((c) => {
   185→      if (c.code === '034') return kajiadoCounty;
   186→      // Inject real audit opinion from county-audit-data
   187→      const auditRecords = getCountyAuditRecords(c.code);
   188→      const latestAudit = auditRecords.find(a => a.financialYear === 'FY 2024/25');
   189→      const latestBudget = getCountyBudget(c.code, 'FY 2024/25');
   190→      return {
   191→        ...c,
   192→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   193→        executiveAuditSource: latestAudit?.source,
   194→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
   195→        developmentAbsorptionSource: latestBudget ? {
   196→          source: latestBudget.source.source,
   197→          reportTitle: latestBudget.source.reportTitle,
   198→          financialYear: latestBudget.source.financialYear,
   199→          url: latestBudget.source.url,
   200→          accessedDate: latestBudget.source.accessedDate,
   201→        } : undefined,
   202→        dataAvailability: 'partial' as const,
   203→        dataAvailabilityNote: `Audit opinion: ${latestAudit?.executiveOpinion || 'pending'} (FY 2024/25). Budget absorption: ${latestBudget?.devAbsorptionRate ?? 'pending'}%. Full official data from oagkenya.go.ke and cob.go.ke.`,
   204→      };
   205→    });
   206→  }, []);
   207→
   208→  const toggleCounty = (code: string) => {
   209→    const next = new Set(expandedCounties);
   210→    if (next.has(code)) next.delete(code); else next.add(code);
   211→    setExpandedCounties(next);
   212→  };
   213→
   214→  const addToComparison = (rep: Representative, countyName: string) => {
   215→    if (comparisonList.length >= 4) return;
   216→    if (comparisonList.find(c => c.representative.id === rep.id)) return;
   217→    setComparisonList([...comparisonList, { representative: rep, countyName }]);
   218→  };
   219→
   220→  const removeFromComparison = (id: string) => {
   221→    setComparisonList(comparisonList.filter(c => c.representative.id !== id));
   222→  };
   223→
   224→  const filteredGovernors = useMemo(() => {
   225→    return all47Governors.filter((g) => {
   226→      if (filters.region && g.region !== filters.region) return false;
   227→      if (filters.coalition && g.coalition !== filters.coalition) return false;
   228→      if (filters.party && g.party !== filters.party) return false;
   229→      if (filters.keyword) {
   230→        const kw = filters.keyword.toLowerCase();
   231→        if (!g.name.toLowerCase().includes(kw) && !g.county.toLowerCase().includes(kw)) return false;
   232→      }
   233→      return true;
   234→    });
   235→  }, [filters]);
   236→
   237→  // Group nav items by section
   238→  const governanceItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.governance');
   239→  const civicItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.civicTools');
   240→  const citizenItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.citizenAction');
   241→  const dataItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.dataAlerts');
   242→  const analyticsItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.analytics');
   243→  const leadershipItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.leadershipProjects');
   244→  const insightItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.insights');
   245→
   246→  // Section labels lookup
   247→  const sectionLabels = useMemo(() => ({
   248→    'nav.sections.governance': t('nav.sections.governance'),
   249→    'nav.sections.civicTools': t('nav.sections.civicTools'),
   250→    'nav.sections.citizenAction': t('nav.sections.citizenAction'),
   251→    'nav.sections.dataAlerts': t('nav.sections.dataAlerts'),
   252→    'nav.sections.analytics': t('nav.sections.analytics'),
   253→    'nav.sections.leadershipProjects': t('nav.sections.leadershipProjects'),
   254→    'nav.sections.insights': t('nav.sections.insights'),
   255→  }), [t]);
   256→
   257→  // Helper to get localized label for a nav item
   258→  const getItemLabel = (id: TabId) => navItems.find(n => n.id === id)?.label ?? id;
   259→
   260→  // Helper to get localized section label
   261→  const getSectionLabel = (sectionKey: string) => sectionLabels[sectionKey] ?? sectionKey;
   262→
   263→  // Sidebar section renderer helper
   264→  const renderSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   265→    <div>
   266→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   267→      {items.map((item) => (
   268→        <button
   269→          key={item.id}
   270→          onClick={() => setActiveTab(item.id)}
   271→          aria-current={activeTab === item.id ? 'page' : undefined}
   272→          className={`w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-xs font-medium transition-colors mb-0.5 ${
   273→            activeTab === item.id
   274→              ? `${activeColor} text-white shadow-sm`
   275→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   276→          }`}
   277→        >
   278→          <item.icon className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   279→          <span>{getItemLabel(item.id)}</span>
   280→        </button>
   281→      ))}
   282→    </div>
   283→  );
   284→
   285→  // Mobile sidebar section renderer helper
   286→  const renderMobileSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   287→    <div>
   288→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   289→      {items.map((item) => (
   290→        <button
   291→          key={item.id}
   292→          onClick={() => { setActiveTab(item.id); setSidebarOpen(false); }}
   293→          aria-current={activeTab === item.id ? 'page' : undefined}
   294→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors mb-0.5 ${
   295→            activeTab === item.id
   296→              ? `${activeColor} text-white`
   297→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   298→          }`}
   299→        >
   300→          <item.icon className="h-4 w-4" aria-hidden="true" />
   301→          <span>{getItemLabel(item.id)}</span>
   302→        </button>
   303→      ))}
   304→    </div>
   305→  );
   306→
   307→  const { theme, setTheme } = useTheme();
   308→
   309→  // Keyboard shortcuts
   310→  React.useEffect(() => {
   311→    const handler = (e: KeyboardEvent) => {
   312→      if (e.key === 'Escape') { setSidebarOpen(false); return; }
   313→      if ((e.ctrlKey || e.metaKey) && e.key === 'k') { e.preventDefault(); commandPaletteRef.current?.open(); return; }
   314→      if ((e.ctrlKey || e.metaKey) && e.key === 'd') { e.preventDefault(); setTheme(theme === 'dark' ? 'light' : 'dark'); return; }
   315→      if (e.key === '?' && !['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) {
   316→        alert('Keyboard Shortcuts:\nCtrl+K: Search\nCtrl+D: Toggle Dark Mode\nEsc: Close panels\n?: This help');
   317→      }
   318→    };
   319→    window.addEventListener('keydown', handler);
   320→    return () => window.removeEventListener('keydown', handler);
   321→  }, [theme, setTheme]);
   322→
   323→  return (
   324→    <TooltipProvider>
   325→      <div className="min-h-screen flex flex-col bg-stone-50 dark:bg-stone-950">
   326→        {/* ══════════ SKIP NAVIGATION ══════════ */}
   327→        <a href="#main-content" className="skip-to-content">
   328→          Skip to main content
   329→        </a>
   330→
   331→        {/* ══════════ HEADER ══════════ */}
   332→        <header className="bg-white dark:bg-stone-900 border-b border-stone-200 dark:border-stone-700 sticky top-0 z-50">
   333→          <div className="max-w-[1400px] mx-auto px-4 sm:px-6 lg:px-8">
   334→            <div className="py-3 flex items-center justify-between gap-4">
   335→              <div className="flex items-center gap-3">
   336→                <button
   337→                  onClick={() => setSidebarOpen(!sidebarOpen)}
   338→                  aria-label="Open navigation menu"
   339→                  className="lg:hidden h-9 w-9 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 transition-colors"
   340→                >
   341→                  <Menu className="h-4 w-4 text-stone-600 dark:text-stone-300" aria-hidden="true" />
   342→                </button>
   343→                <div className="h-8 w-8 rounded-lg bg-emerald-600 flex items-center justify-center shrink-0" aria-hidden="true">
   344→                  <Shield className="h-4 w-4 text-white" />
   345→                </div>
   346→                <div>
   347→                  <h1 className="text-sm sm:text-base font-bold text-stone-900 dark:text-stone-100 tracking-tight leading-tight">
   348→                    {t('app.title')}
   349→                  </h1>
   350→                  <p className="text-[10px] sm:text-xs text-stone-500">
   351→                    {t('app.subtitle')}
   352→                  </p>
   353→                </div>
   354→              </div>
   355→              <div className="flex items-center gap-2">
   356→                {comparisonList.length > 0 && (
   357→                  <Badge variant="secondary" className="text-[10px] h-6 bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300">
   358→                    {t('app.headerCompare', { count: comparisonList.length })}
   359→                  </Badge>
   360→                )}
   361→                <button
   362→                  onClick={() => commandPaletteRef.current?.open()}
   363→                  aria-label={t('app.searchShortcut')}
   364→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   365→                >
   366→                  <Keyboard className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" />
   367→                </button>
   368→                <button
   369→                  onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   370→                  aria-label={t('app.darkModeShortcut')}
   371→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   372→                >
   373→                  {theme === 'dark' ? <Sun className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" /> : <Moon className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" />}
   374→                </button>
   375→                <LanguageToggle />
   376→                <a href="https://kenyalaw.org/" target="_blank" rel="noopener noreferrer" className="hidden sm:flex items-center gap-1 text-[10px] text-stone-500 hover:text-emerald-600 transition-colors">
