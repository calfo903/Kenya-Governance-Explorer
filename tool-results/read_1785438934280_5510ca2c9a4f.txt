   145→// ─── Helper: interpolate color from red to green ────────────────────
   146→
   147→function lerpColor(t: number): string {
   148→  const clamped = Math.max(0, Math.min(1, t));
   149→  const r = Math.round(220 * (1 - clamped) + 22 * clamped);
   150→  const g = Math.round(38 * (1 - clamped) + 163 * clamped);
   151→  const b = Math.round(38 * (1 - clamped) + 74 * clamped);
   152→  return `rgb(${r}, ${g}, ${b})`;
   153→}
   154→
   155→// ─── Helper: population density color ───────────────────────────────
   156→
   157→function densityColor(pop: number, area: number): string {
   158→  const density = pop / area;
   159→  if (density > 2000) return "#7f1d1d";
   160→  if (density > 1000) return "#991b1b";
   161→  if (density > 500) return "#b91c1c";
   162→  if (density > 200) return "#c2410c";
   163→  if (density > 100) return "#ea580c";
   164→  if (density > 50) return "#f59e0b";
   165→  if (density > 20) return "#fbbf24";
   166→  return "#fef08a";
   167→}

   177→function buildAuditMap(fy?: string) {
   178→  const m: Record<string, (typeof countyAuditData)[number]> = {};
   179→  for (const r of countyAuditData) {
   180→    if (fy) {
   181→      // Filter to specific financial year
   182→      if (r.financialYear === fy) {
   183→        m[r.countyCode] = r;
   184→      }
   185→    } else {
   186→      // Pick latest
   187→      if (!m[r.countyCode] || r.financialYear > m[r.countyCode].financialYear) {
   188→        m[r.countyCode] = r;
   189→      }
   190→    }
   191→  }
   192→  return m;
   193→}
   194→
   195→function buildBudgetAbsorptionMap(fy?: string) {
   196→  const m: Record<string, number> = {};
   197→  if (fy) {
   198→    // Filter to specific financial year, prefer full year records
   199→    for (const r of countyBudgetData) {
   200→      if (r.financialYear === fy && r.period === "Full Year") {
   201→        m[r.countyCode] = r.devAbsorptionRate;
   202→      }
   203→    }
   204→    // Fall back to any record in that FY if no full year record
   205→    for (const r of countyBudgetData) {
   206→      if (r.financialYear === fy && !m[r.countyCode]) {
   207→        m[r.countyCode] = r.devAbsorptionRate;
   208→      }
   209→    }
   210→  } else {
   211→    // Default: pick FY 2024/25 full year records
   212→    for (const r of countyBudgetData) {
   213→      if (!m[r.countyCode] || r.financialYear > "FY 2024/24") {
   214→        m[r.countyCode] = r.devAbsorptionRate;
   215→      }
   216→    }
   217→    // Ensure we pick the FY 2024/25 full year records
   218→    for (const r of countyBudgetData) {
   219→      if (r.financialYear === "FY 2024/25" && r.period === "Full Year") {
   220→        m[r.countyCode] = r.devAbsorptionRate;
   221→      }
   222→    }
   223→  }
   224→  return m;
   225→}

   317→  return "bg-red-500";
   318→}

   384→// ─── County color resolver ──────────────────────────────────────────
   385→
   386→function resolveCountyFill(
   387→  code: string,
   388→  mode: ColorMode,
   389→  govMap: Record<string, (typeof all47Governors)[number]>,
   390→  auditMap: Record<string, (typeof countyAuditData)[number]>,
   391→  budgetMap: Record<string, number>,
   392→): string {
   393→  switch (mode) {
   394→    case "coalition": {
   395→      const gov = govMap[code];
   396→      return gov ? (COALITION_COLORS[gov.coalition] ?? DEFAULT_FILL) : DEFAULT_FILL;
   397→    }
   398→    case "region": {
   399→      const gov = govMap[code];
   400→      return gov ? (REGION_COLORS[gov.region] ?? DEFAULT_FILL) : DEFAULT_FILL;
   401→    }
   402→    case "audit": {
   403→      const audit = auditMap[code];
   404→      if (!audit || !audit.executiveOpinion) return DEFAULT_FILL;
   405→      return AUDIT_COLORS[audit.executiveOpinion] ?? DEFAULT_FILL;
   406→    }
   407→    case "budget": {
   408→      const rate = budgetMap[code];
   409→      if (rate == null) return DEFAULT_FILL;
   410→      return lerpColor(rate / 100);
   411→    }
   412→    case "population": {
   413→      const gov = govMap[code];
   414→      if (!gov) return DEFAULT_FILL;
   415→      return densityColor(gov.population, gov.areaSqKm);
   416→    }
   417→    default:
   418→      return DEFAULT_FILL;
   419→  }
   420→}
