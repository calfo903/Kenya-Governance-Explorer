   158→   158→function getInitials(name: string): string {
   159→   159→  const cleaned = name.replace(/^(Hon\.?|Dr\.?|Maj\.?|Eng\.?|Prof\.?)\s*/i, '');
   160→   160→  const parts = cleaned.split(/\s+/).filter(Boolean);
   161→   161→  if (parts.length >= 2) return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   162→   162→  return parts[0] ? parts[0].substring(0, 2).toUpperCase() : '??';
   163→   163→}
   164→   164→
   165→   165→function formatTerm(start: string, end: string): string {
   166→   166→  const fmt = (d: string) => {
   167→   167→    const date = new Date(d);
   168→   168→    return date.toLocaleDateString('en-KE', { month: 'short', year: 'numeric' });
   169→   169→  };
   170→   170→  return `${fmt(start)} – ${fmt(end)}`;
   171→   171→}
   172→   172→
   173→   173→function shortenCoalition(coalition: string): string {
   174→   174→  if (coalition.includes('Kenya Kwanza')) return 'Kenya Kwanza';
   175→   175→  if (coalition.includes('Azimio')) return 'Azimio';
   176→   176→  if (coalition === 'Independent') return 'Independent';
   177→   177→  return coalition.length > 30 ? coalition.substring(0, 28) + '…' : coalition;
   178→   178→}
