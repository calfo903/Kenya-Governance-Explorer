   158→function getInitials(name: string): string {
   159→  const cleaned = name.replace(/^(Hon\.?|Dr\.?|Maj\.?|Eng\.?|Prof\.?)\s*/i, '');
   160→  const parts = cleaned.split(/\s+/).filter(Boolean);
   161→  if (parts.length >= 2) return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   162→  return parts[0] ? parts[0].substring(0, 2).toUpperCase() : '??';
   163→}
   164→
   165→function formatTerm(start: string, end: string): string {
   166→  const fmt = (d: string) => {
   167→    const date = new Date(d);
   168→    return date.toLocaleDateString('en-KE', { month: 'short', year: 'numeric' });
   169→  };
   170→  return `${fmt(start)} – ${fmt(end)}`;
   171→}
   172→
   173→function shortenCoalition(coalition: string): string {
   174→  if (coalition.includes('Kenya Kwanza')) return 'Kenya Kwanza';
   175→  if (coalition.includes('Azimio')) return 'Azimio';
   176→  if (coalition === 'Independent') return 'Independent';
   177→  return coalition.length > 30 ? coalition.substring(0, 28) + '…' : coalition;
   178→}
