   166→
   167→function PageContent() {
   168→  const t = useTranslations();
   169→  const locale = useLocaleStore((s) => s.locale);
   170→  const [activeTab, setActiveTab] = useState<TabId>('summary');
   171→  const [selectedCounty, setSelectedCounty] = useState<string>('034');
   172→  const [expandedCounties, setExpandedCounties] = useState<Set<string>>(new Set());
   173→  const [comparisonList, setComparisonList] = useState<ComparisonItem[]>([]);
   174→  const [filters, setFilters] = useState<FilterState>({});
   175→  const [sidebarOpen, setSidebarOpen] = useState(false);
   176→  const commandPaletteRef = React.useRef<{ open: () => void; close: () => void }>(null);
   177→
   178→  // Localized nav items
   179→  const navItems = useMemo(() => navItemDefs.map(def => ({
   180→    ...def,
   181→    label: t(def.labelKey),
   182→  })), [t]);
   183→
   184→  const allCounties = useMemo((): County[] => {
   185→    const placeholders = getPlaceholderCounties();
   186→    return placeholders.map((c) => {
   187→      if (c.code === '034') return kajiadoCounty;
   188→      // Inject real audit opinion from county-audit-data
   189→      const auditRecords = getCountyAuditRecords(c.code);
   190→      const latestAudit = auditRecords.find(a => a.financialYear === 'FY 2024/25');
   191→      const latestBudget = getCountyBudget(c.code, 'FY 2024/25');
   192→      return {
   193→        ...c,
   194→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   195→        executiveAuditSource: latestAudit?.source,
   196→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
   197→        developmentAbsorptionSource: latestBudget ? {
   198→          source: latestBudget.source.source,
   199→          reportTitle: latestBudget.source.reportTitle,
   200→          financialYear: latestBudget.source.financialYear,
   201→          url: latestBudget.source.url,
   202→          accessedDate: latestBudget.source.accessedDate,
   203→        } : undefined,
   204→        dataAvailability: 'partial' as const,
   205→        dataAvailabilityNote: `Audit opinion: ${latestAudit?.executiveOpinion || 'pending'} (FY 2024/25). Budget absorption: ${latestBudget?.devAbsorptionRate ?? 'pending'}%. Full official data from oagkenya.go.ke and cob.go.ke.`,
   206→      };
   207→    });
   208→  }, []);
   209→
   210→  const toggleCounty = (code: string) => {
   211→    const next = new Set(expandedCounties);
   212→    if (next.has(code)) next.delete(code); else next.add(code);
   213→    setExpandedCounties(next);
   214→  };
   215→
   216→  const addToComparison = (rep: Representative, countyName: string) => {
   217→    if (comparisonList.length >= 4) return;
   218→    if (comparisonList.find(c => c.representative.id === rep.id)) return;
   219→    setComparisonList([...comparisonList, { representative: rep, countyName }]);
   220→  };
   221→
   222→  const removeFromComparison = (id: string) => {
   223→    setComparisonList(comparisonList.filter(c => c.representative.id !== id));
   224→  };
   225→
   226→  const filteredGovernors = useMemo(() => {
   227→    return all47Governors.filter((g) => {
   228→      if (filters.region && g.region !== filters.region) return false;
   229→      if (filters.coalition && g.coalition !== filters.coalition) return false;
   230→      if (filters.party && g.party !== filters.party) return false;
   231→      if (filters.keyword) {
   232→        const kw = filters.keyword.toLowerCase();
   233→        if (!g.name.toLowerCase().includes(kw) && !g.county.toLowerCase().includes(kw)) return false;
   234→      }
   235→      return true;
   236→    });
   237→  }, [filters]);
   238→
   239→  // Group nav items by section
   240→  const governanceItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.governance');
   241→  const civicItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.civicTools');
   242→  const citizenItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.citizenAction');
   243→  const dataItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.dataAlerts');
   244→  const analyticsItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.analytics');
   245→  const leadershipItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.leadershipProjects');
   246→  const insightItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.insights');
   247→
   248→  // Section labels lookup
   249→  const sectionLabels = useMemo(() => ({
   250→    'nav.sections.governance': t('nav.sections.governance'),
   251→    'nav.sections.civicTools': t('nav.sections.civicTools'),
   252→    'nav.sections.citizenAction': t('nav.sections.citizenAction'),
   253→    'nav.sections.dataAlerts': t('nav.sections.dataAlerts'),
   254→    'nav.sections.analytics': t('nav.sections.analytics'),
   255→    'nav.sections.leadershipProjects': t('nav.sections.leadershipProjects'),
   256→    'nav.sections.insights': t('nav.sections.insights'),
   257→  }), [t]);
   258→
   259→  // Helper to get localized label for a nav item
   260→  const getItemLabel = (id: TabId) => navItems.find(n => n.id === id)?.label ?? id;
   261→
   262→  // Helper to get localized section label
   263→  const getSectionLabel = (sectionKey: string) => sectionLabels[sectionKey] ?? sectionKey;
   264→
   265→  // Sidebar section renderer helper
   266→  const renderSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   267→    <div>
   268→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   269→      {items.map((item) => (
   270→        <button
   271→          key={item.id}
   272→          onClick={() => setActiveTab(item.id)}
   273→          aria-current={activeTab === item.id ? 'page' : undefined}
   274→          className={`w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-xs font-medium transition-colors mb-0.5 ${
   275→            activeTab === item.id
   276→              ? `${activeColor} text-white shadow-sm`
   277→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   278→          }`}
   279→        >
   280→          <item.icon className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   281→          <span>{getItemLabel(item.id)}</span>
   282→        </button>
   283→      ))}
   284→    </div>
   285→  );
   286→
   287→  // Mobile sidebar section renderer helper
   288→  const renderMobileSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   289→    <div>
   290→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   291→      {items.map((item) => (
   292→        <button
   293→          key={item.id}
   294→          onClick={() => { setActiveTab(item.id); setSidebarOpen(false); }}
   295→          aria-current={activeTab === item.id ? 'page' : undefined}
   296→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors mb-0.5 ${
   297→            activeTab === item.id
   298→              ? `${activeColor} text-white`
   299→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   300→          }`}
   301→        >
   302→          <item.icon className="h-4 w-4" aria-hidden="true" />
   303→          <span>{getItemLabel(item.id)}</span>
   304→        </button>
   305→      ))}
   306→    </div>
   307→  );
   308→
   309→  const { theme, setTheme } = useTheme();
   310→
   311→  // Keyboard shortcuts
   312→  React.useEffect(() => {
   313→    const handler = (e: KeyboardEvent) => {
   314→      if (e.key === 'Escape') { setSidebarOpen(false); return; }
   315→      if ((e.ctrlKey || e.metaKey) && e.key === 'k') { e.preventDefault(); commandPaletteRef.current?.open(); return; }
   316→      if ((e.ctrlKey || e.metaKey) && e.key === 'd') { e.preventDefault(); setTheme(theme === 'dark' ? 'light' : 'dark'); return; }
   317→      if (e.key === '?' && !['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) {
   318→        alert('Keyboard Shortcuts:\nCtrl+K: Search\nCtrl+D: Toggle Dark Mode\nEsc: Close panels\n?: This help');
   319→      }
   320→    };
   321→    window.addEventListener('keydown', handler);
   322→    return () => window.removeEventListener('keydown', handler);
   323→  }, [theme, setTheme]);
   324→
   325→  return (
   326→    <TooltipProvider>
   327→      <div className="min-h-screen flex flex-col bg-stone-50 dark:bg-stone-950">
   328→        {/* ══════════ SKIP NAVIGATION ══════════ */}
   329→        <a href="#main-content" className="skip-to-content">
   330→          Skip to main content
   331→        </a>
   332→
   333→        {/* ══════════ HEADER ══════════ */}
   334→        <header className="bg-white dark:bg-stone-900 border-b border-stone-200 dark:border-stone-700 sticky top-0 z-50">
   335→          <div className="max-w-[1400px] mx-auto px-4 sm:px-6 lg:px-8">
   336→            <div className="py-3 flex items-center justify-between gap-4">
   337→              <div className="flex items-center gap-3">
   338→                <button
   339→                  onClick={() => setSidebarOpen(!sidebarOpen)}
   340→                  aria-label="Open navigation menu"
   341→                  className="lg:hidden h-9 w-9 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 transition-colors"
   342→                >
   343→                  <Menu className="h-4 w-4 text-stone-600 dark:text-stone-300" aria-hidden="true" />
   344→                </button>
   345→                <div className="h-8 w-8 rounded-lg bg-emerald-600 flex items-center justify-center shrink-0" aria-hidden="true">
   346→                  <Shield className="h-4 w-4 text-white" />
   347→                </div>
   348→                <div>
   349→                  <h1 className="text-sm sm:text-base font-bold text-stone-900 dark:text-stone-100 tracking-tight leading-tight">
   350→                    {t('app.title')}
   351→                  </h1>
   352→                  <p className="text-[10px] sm:text-xs text-stone-500">
   353→                    {t('app.subtitle')}
   354→                  </p>
   355→                </div>
   356→              </div>
   357→              <div className="flex items-center gap-2">
   358→                {comparisonList.length > 0 && (
   359→                  <Badge variant="secondary" className="text-[10px] h-6 bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300">
   360→                    {t('app.headerCompare', { count: comparisonList.length })}
   361→                  </Badge>
   362→                )}
   363→                <button
   364→                  onClick={() => commandPaletteRef.current?.open()}
   365→                  aria-label={t('app.searchShortcut')}
   366→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   367→                >
   368→                  <Keyboard className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" />
   369→                </button>
   370→                <button
   371→                  onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   372→                  aria-label={t('app.darkModeShortcut')}
   373→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   374→                >
   375→                  {theme === 'dark' ? <Sun className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" /> : <Moon className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" />}
