   171→function PageContent() {
   172→  const t = useTranslations();
   173→  const locale = useLocaleStore((s) => s.locale);

   178→  const [filters, setFilters] = useState<FilterState>({});
   179→  const [sidebarOpen, setSidebarOpen] = useState(false);
   180→  const commandPaletteRef = React.useRef<{ open: () => void; close: () => void }>(null);
   181→
   182→  // Localized nav items
   183→  const navItems = useMemo(() => navItemDefs.map(def => ({
   184→    ...def,
   185→    label: t(def.labelKey),
   186→  })), [t]);
   187→
   188→  const allCounties = useMemo((): County[] => {
   189→    const placeholders = getPlaceholderCounties();
   190→    return placeholders.map((c) => {
   191→      if (c.code === '034') return kajiadoCounty;
   192→      // Inject real audit opinion from county-audit-data
   193→      const auditRecords = getCountyAuditRecords(c.code);
   194→      const latestAudit = auditRecords.find(a => a.financialYear === 'FY 2024/25');
   195→      const latestBudget = getCountyBudget(c.code, 'FY 2024/25');
   196→      return {
   197→        ...c,
   198→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   199→        executiveAuditSource: latestAudit?.source,
   200→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
   201→        developmentAbsorptionSource: latestBudget ? {
   202→          source: latestBudget.source.source,
   203→          reportTitle: latestBudget.source.reportTitle,
   204→          financialYear: latestBudget.source.financialYear,
   205→          url: latestBudget.source.url,
   206→          accessedDate: latestBudget.source.accessedDate,
   207→        } : undefined,
   208→        dataAvailability: 'partial' as const,
   209→        dataAvailabilityNote: `Audit opinion: ${latestAudit?.executiveOpinion || 'pending'} (FY 2024/25). Budget absorption: ${latestBudget?.devAbsorptionRate ?? 'pending'}%. Full official data from oagkenya.go.ke and cob.go.ke.`,
   210→      };
   211→    });
   212→  }, []);
   213→
   214→  const toggleCounty = (code: string) => {
   215→    const next = new Set(expandedCounties);
   216→    if (next.has(code)) next.delete(code); else next.add(code);

   218→  };

   221→    if (comparisonList.length >= 4) return;
   222→    if (comparisonList.find(c => c.representative.id === rep.id)) return;
   223→    setComparisonList([...comparisonList, { representative: rep, countyName }]);
   224→  };
   225→
   226→  const removeFromComparison = (id: string) => {
   227→    setComparisonList(comparisonList.filter(c => c.representative.id !== id));
   228→  };
   229→
   230→  const filteredGovernors = useMemo(() => {
   231→    return all47Governors.filter((g) => {
   232→      if (filters.region && g.region !== filters.region) return false;
   233→      if (filters.coalition && g.coalition !== filters.coalition) return false;
   234→      if (filters.party && g.party !== filters.party) return false;
   235→      if (filters.keyword) {
   236→        const kw = filters.keyword.toLowerCase();
   237→        if (!g.name.toLowerCase().includes(kw) && !g.county.toLowerCase().includes(kw)) return false;
   238→      }
   239→      return true;
   240→    });
   241→  }, [filters]);
   242→
   243→  // Group nav items by section
   244→  const governanceItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.governance');
   245→  const civicItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.civicTools');
   246→  const citizenItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.citizenAction');
   247→  const dataItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.dataAlerts');

   250→  const insightItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.insights');

   254→    'nav.sections.governance': t('nav.sections.governance'),
   255→    'nav.sections.civicTools': t('nav.sections.civicTools'),

   259→    'nav.sections.leadershipProjects': t('nav.sections.leadershipProjects'),
   260→    'nav.sections.insights': t('nav.sections.insights'),
   261→  }), [t]);

   265→
   266→  // Helper to get localized section label
   267→  const getSectionLabel = (sectionKey: string) => sectionLabels[sectionKey] ?? sectionKey;
   268→

   282→          }`}
   283→        >
   284→          <item.icon className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   285→          <span>{getItemLabel(item.id)}</span>
   286→        </button>
   287→      ))}

   292→  const renderMobileSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (

   294→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>

   296→        <button
   297→          key={item.id}
   298→          onClick={() => { setActiveTab(item.id); setSidebarOpen(false); }}

   301→            activeTab === item.id
   302→              ? `${activeColor} text-white`
   303→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   304→          }`}
   305→        >
   306→          <item.icon className="h-4 w-4" aria-hidden="true" />
   307→          <span>{getItemLabel(item.id)}</span>
   308→        </button>
   309→      ))}
   310→    </div>
   311→  );
   312→
   313→  const { theme, setTheme } = useTheme();
   314→
   315→  // Keyboard shortcuts
   316→  React.useEffect(() => {
   317→    const handler = (e: KeyboardEvent) => {
   318→      if (e.key === 'Escape') { setSidebarOpen(false); return; }

   320→      if ((e.ctrlKey || e.metaKey) && e.key === 'd') { e.preventDefault(); setTheme(theme === 'dark' ? 'light' : 'dark'); return; }
   321→      if (e.key === '?' && !['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) {
   322→        alert('Keyboard Shortcuts:\nCtrl+K: Search\nCtrl+D: Toggle Dark Mode\nEsc: Close panels\n?: This help');
   323→      }
   324→    };
   325→    window.addEventListener('keydown', handler);
   326→    return () => window.removeEventListener('keydown', handler);
   327→  }, [theme, setTheme]);
   328→
   329→  return (

   331→      <div className="min-h-screen flex flex-col bg-stone-50 dark:bg-stone-950">
   332→        {/* ══════════ SKIP NAVIGATION ══════════ */}

   334→          Skip to main content
   335→        </a>
   336→
   337→        {/* ══════════ HEADER ══════════ */}
   338→        <header className="bg-white dark:bg-stone-900 border-b border-stone-200 dark:border-stone-700 sticky top-0 z-50">
   339→          <div className="max-w-[1400px] mx-auto px-4 sm:px-6 lg:px-8">
   340→            <div className="py-3 flex items-center justify-between gap-4">

   343→                  onClick={() => setSidebarOpen(!sidebarOpen)}
   344→                  aria-label="Open navigation menu"
   345→                  className="lg:hidden h-9 w-9 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 transition-colors"
   346→                >
   347→                  <Menu className="h-4 w-4 text-stone-600 dark:text-stone-300" aria-hidden="true" />
   348→                </button>
   349→                <div className="h-8 w-8 rounded-lg bg-emerald-600 flex items-center justify-center shrink-0" aria-hidden="true">
   350→                  <Shield className="h-4 w-4 text-white" />
   351→                </div>
   352→                <div>
   353→                  <h1 className="text-sm sm:text-base font-bold text-stone-900 dark:text-stone-100 tracking-tight leading-tight">

   357→                    {t('app.subtitle')}

   365→                  </Badge>

   367→                <button
   368→                  onClick={() => commandPaletteRef.current?.open()}
   369→                  aria-label={t('app.searchShortcut')}
   370→                  className="h-8 w-8 rounded-lg border border-stone-200 dark:border-stone-700 flex items-center justify-center hover:bg-stone-50 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 dark:bg-stone-800 dark:hover:bg-stone-800 transition-colors"
   371→                >
   372→                  <Keyboard className="h-3.5 w-3.5 text-stone-600 dark:text-stone-300" aria-hidden="true" />
   373→                </button>
   374→                <button
   375→                  onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   376→                  aria-label={t('app.darkModeShortcut')}
