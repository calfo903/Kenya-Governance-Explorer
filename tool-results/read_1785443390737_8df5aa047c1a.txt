   157→    const onKeyDown = (e: KeyboardEvent) => { if (e.key === 'Shift') shiftHeldRef.current = true; };
   158→    const onKeyUp = (e: KeyboardEvent) => { if (e.key === 'Shift') shiftHeldRef.current = false; };
   159→    window.addEventListener('keydown', onKeyDown);
   160→    window.addEventListener('keyup', onKeyUp);
   161→    return () => {
   162→      window.removeEventListener('keydown', onKeyDown);
   163→      window.removeEventListener('keyup', onKeyUp);
   164→    };

   166→
   167→  const governorMap = useMemo(() => {
   168→    const map = new Map<string, GovernorEntry>();

   171→  }, []);
   172→

   174→    const gov = governorMap.get(shape.name);
   175→    if (colorMode === 'coalition') return gov ? COALITION_COLORS[gov.coalition] || '#9ca3af' : '#9ca3af';
   176→    if (colorMode === 'region') return gov ? REGION_COLORS[gov.region] || '#9ca3af' : '#9ca3af';
   177→    if (colorMode === 'audit') {
   178→      const auditKey = SIMULATED_AUDIT[shape.name] || 'qualified';
   179→      return AUDIT_COLORS[auditKey];
   180→    }
   181→    return '#9ca3af';
   182→  };
   183→
   184→  const filteredShapes = useMemo(() => {
   185→    let shapes = countyShapes;
   186→    if (searchQuery) {
   187→      const q = searchQuery.toLowerCase();
   188→      shapes = shapes.filter(s => s.name.toLowerCase().includes(q));
   189→    }
   190→    if (regionFilter !== 'all') {
   191→      shapes = shapes.filter(s => governorMap.get(s.name)?.region === regionFilter);
   192→    }
   193→    if (coalitionFilter !== 'all' && colorMode === 'coalition') {
   194→      shapes = shapes.filter(s => governorMap.get(s.name)?.coalition === coalitionFilter);
   195→    }
   196→    if (auditFilter !== 'all' && colorMode === 'audit') {
   197→      shapes = shapes.filter(s => SIMULATED_AUDIT[s.name] === auditFilter);
   198→    }
   199→    // Apply quick filter
   200→    if (quickFilter === 'adverse') {
   201→      shapes = shapes.filter(s => {
   202→        const rec = countyAuditData.find(a => a.countyCode === s.code && a.financialYear === selectedFY);
   203→        return rec && rec.executiveOpinion === 'Adverse';
   204→      });
   205→    } else if (quickFilter === 'top5-budget') {
   206→      const budgetForFY = countyBudgetData
   207→        .filter(b => b.financialYear === selectedFY)
   208→        .sort((a, b) => b.devAbsorptionRate - a.devAbsorptionRate)
   209→        .slice(0, 5)
   210→        .map(b => b.countyCode);
   211→      shapes = shapes.filter(s => budgetForFY.includes(s.code));
   212→    } else if (quickFilter === 'largest-pop') {
   213→      const top10 = [...all47Governors].sort((a, b) => b.population - a.population).slice(0, 10).map(g => g.code);
   214→      shapes = shapes.filter(s => top10.includes(s.code));
   215→    }
   216→    return shapes;
   217→  }, [regionFilter, coalitionFilter, auditFilter, searchQuery, colorMode, governorMap, quickFilter, selectedFY]);

   222→    return {
   223→      'adverse': adverseCount,
   224→      'top5-budget': 5,
   225→      'largest-pop': 10,
   226→    };

   244→        prev.includes(code) ? prev.filter(c => c !== code) : [...prev, code]
   245→      );

   247→      const gov = all47Governors.find(g => g.code === code);
   248→      if (gov) setSelectedCounty(gov);
   249→    }
   250→  }, []);

   252→  const removeMultiSelected = useCallback((code: string) => {
   253→    setMultiSelectedCounties(prev => prev.filter(c => c !== code));
   254→  }, []);

   260→        <div className="flex items-start gap-4">
