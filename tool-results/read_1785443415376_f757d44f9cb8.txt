   146→// ─── Helper: interpolate color from red to green ────────────────────
   147→
   148→function lerpColor(t: number): string {
   149→  const clamped = Math.max(0, Math.min(1, t));
   150→  const r = Math.round(220 * (1 - clamped) + 22 * clamped);
   151→  const g = Math.round(38 * (1 - clamped) + 163 * clamped);
   152→  const b = Math.round(38 * (1 - clamped) + 74 * clamped);
   153→  return `rgb(${r}, ${g}, ${b})`;
   154→}

   156→// ─── Helper: population density color ───────────────────────────────
   157→
   158→function densityColor(pop: number, area: number): string {
   159→  const density = pop / area;
   160→  if (density > 2000) return "#7f1d1d";
   161→  if (density > 1000) return "#991b1b";
   162→  if (density > 500) return "#b91c1c";
   163→  if (density > 200) return "#c2410c";
   164→  if (density > 100) return "#ea580c";
   165→  if (density > 50) return "#f59e0b";
   166→  if (density > 20) return "#fbbf24";
   167→  return "#fef08a";
   168→}

   178→function buildAuditMap(fy?: string) {
   179→  const m: Record<string, (typeof countyAuditData)[number]> = {};
   180→  for (const r of countyAuditData) {
   181→    if (fy) {
   182→      // Filter to specific financial year
   183→      if (r.financialYear === fy) {
   184→        m[r.countyCode] = r;
   185→      }
   186→    } else {
   187→      // Pick latest
   188→      if (!m[r.countyCode] || r.financialYear > m[r.countyCode].financialYear) {
   189→        m[r.countyCode] = r;
   190→      }
   191→    }
   192→  }
   193→  return m;
   194→}
   195→
   196→function buildBudgetAbsorptionMap(fy?: string) {
   197→  const m: Record<string, number> = {};
   198→  if (fy) {
   199→    // Filter to specific financial year, prefer full year records
   200→    for (const r of countyBudgetData) {
   201→      if (r.financialYear === fy && r.period === "Full Year") {
   202→        m[r.countyCode] = r.devAbsorptionRate;
   203→      }
   204→    }
   205→    // Fall back to any record in that FY if no full year record
   206→    for (const r of countyBudgetData) {
   207→      if (r.financialYear === fy && !m[r.countyCode]) {
   208→        m[r.countyCode] = r.devAbsorptionRate;
   209→      }
   210→    }
   211→  } else {
   212→    // Default: pick FY 2024/25 full year records
   213→    for (const r of countyBudgetData) {
   214→      if (!m[r.countyCode] || r.financialYear > "FY 2024/24") {
   215→        m[r.countyCode] = r.devAbsorptionRate;
   216→      }
   217→    }
   218→    // Ensure we pick the FY 2024/25 full year records
   219→    for (const r of countyBudgetData) {
   220→      if (r.financialYear === "FY 2024/25" && r.period === "Full Year") {
   221→        m[r.countyCode] = r.devAbsorptionRate;
   222→      }
   223→    }
   224→  }
   225→  return m;
   226→}

   418→// ─── County color resolver ──────────────────────────────────────────
   419→
   420→function resolveCountyFill(
   421→  code: string,
   422→  mode: ColorMode,
   423→  govMap: Record<string, (typeof all47Governors)[number]>,
   424→  auditMap: Record<string, (typeof countyAuditData)[number]>,
   425→  budgetMap: Record<string, number>,
   426→  cecmScoreMap: Map<string, { overallScore: number }>,
   427→): string {
   428→  switch (mode) {
   429→    case "coalition": {
   430→      const gov = govMap[code];
   431→      return gov ? (COALITION_COLORS[gov.coalition] ?? DEFAULT_FILL) : DEFAULT_FILL;
   432→    }
   433→    case "region": {
   434→      const gov = govMap[code];
   435→      return gov ? (REGION_COLORS[gov.region] ?? DEFAULT_FILL) : DEFAULT_FILL;
   436→    }
   437→    case "audit": {
   438→      const audit = auditMap[code];
   439→      if (!audit || !audit.executiveOpinion) return DEFAULT_FILL;
   440→      return AUDIT_COLORS[audit.executiveOpinion] ?? DEFAULT_FILL;
   441→    }
   442→    case "budget": {
   443→      const rate = budgetMap[code];
   444→      if (rate == null) return DEFAULT_FILL;
   445→      return lerpColor(rate / 100);
   446→    }
   447→    case "cecm-performance": {
   448→      const scoreEntry = cecmScoreMap.get(code);
   449→      if (!scoreEntry) return DEFAULT_FILL;
   450→      return getCECMScoreColor(scoreEntry.overallScore);
   451→    }
   452→    case "population": {
   453→      const gov = govMap[code];
   454→      if (!gov) return DEFAULT_FILL;
   455→      return densityColor(gov.population, gov.areaSqKm);
   456→    }
   457→    default:
   458→      return DEFAULT_FILL;
   459→  }
   460→}
