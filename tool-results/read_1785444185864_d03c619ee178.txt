   197→function PageContent() {
   198→  const t = useTranslations();
   199→  const locale = useLocaleStore((s) => s.locale);
   200→  const [activeTab, setActiveTab] = useState<TabId>('summary');
   201→  const [selectedCounty, setSelectedCounty] = useState<string>('034');
   202→  const [expandedCounties, setExpandedCounties] = useState<Set<string>>(new Set());
   203→  const [comparisonList, setComparisonList] = useState<ComparisonItem[]>([]);
   204→  const [filters, setFilters] = useState<FilterState>({});
   205→  const [sidebarOpen, setSidebarOpen] = useState(false);
   206→  const commandPaletteRef = React.useRef<{ open: () => void; close: () => void }>(null);
   207→
   208→  // Localized nav items
   209→  const navItems = useMemo(() => navItemDefs.map(def => ({
   210→    ...def,
   211→    label: t(def.labelKey),
   212→  })), [t]);
   213→
   214→  const allCounties = useMemo((): County[] => {
   215→    const placeholders = getPlaceholderCounties();
   216→    return placeholders.map((c) => {
   217→      if (c.code === '034') return kajiadoCounty;
   218→      // Inject real audit opinion from county-audit-data
   219→      const auditRecords = getCountyAuditRecords(c.code);
   220→      const latestAudit = auditRecords.find(a => a.financialYear === 'FY 2024/25');
   221→      const latestBudget = getCountyBudget(c.code, 'FY 2024/25');
   222→      return {
   223→        ...c,
   224→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   225→        executiveAuditSource: latestAudit?.source,
   226→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
   227→        developmentAbsorptionSource: latestBudget ? {
   228→          source: latestBudget.source.source,
   229→          reportTitle: latestBudget.source.reportTitle,
   230→          financialYear: latestBudget.source.financialYear,
   231→          url: latestBudget.source.url,
   232→          accessedDate: latestBudget.source.accessedDate,
   233→        } : undefined,
   234→        dataAvailability: 'partial' as const,
   235→        dataAvailabilityNote: `Audit opinion: ${latestAudit?.executiveOpinion || 'pending'} (FY 2024/25). Budget absorption: ${latestBudget?.devAbsorptionRate ?? 'pending'}%. Full official data from oagkenya.go.ke and cob.go.ke.`,
   236→      };
   237→    });
   238→  }, []);
   239→
   240→  const toggleCounty = (code: string) => {
   241→    const next = new Set(expandedCounties);
   242→    if (next.has(code)) next.delete(code); else next.add(code);
   243→    setExpandedCounties(next);
   244→  };
   245→
   246→  const addToComparison = (rep: Representative, countyName: string) => {
   247→    if (comparisonList.length >= 4) return;
   248→    if (comparisonList.find(c => c.representative.id === rep.id)) return;
   249→    setComparisonList([...comparisonList, { representative: rep, countyName }]);
   250→  };
   251→
   252→  const removeFromComparison = (id: string) => {
   253→    setComparisonList(comparisonList.filter(c => c.representative.id !== id));
   254→  };
   255→
   256→  const filteredGovernors = useMemo(() => {
   257→    return all47Governors.filter((g) => {
   258→      if (filters.region && g.region !== filters.region) return false;
   259→      if (filters.coalition && g.coalition !== filters.coalition) return false;
   260→      if (filters.party && g.party !== filters.party) return false;
   261→      if (filters.keyword) {
   262→        const kw = filters.keyword.toLowerCase();
   263→        if (!g.name.toLowerCase().includes(kw) && !g.county.toLowerCase().includes(kw)) return false;
   264→      }
   265→      return true;
   266→    });
   267→  }, [filters]);
   268→
   269→  // Group nav items by section
   270→  const governanceItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.governance');
   271→  const civicItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.civicTools');
   272→  const citizenItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.citizenAction');
   273→  const dataItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.dataAlerts');
   274→  const analyticsItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.analytics');
   275→  const leadershipItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.leadershipProjects');
   276→  const insightItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.insights');
   277→  const aiItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.aiTools');
   278→
   279→  // Section labels lookup
   280→  const sectionLabels = useMemo(() => ({
   281→    'nav.sections.governance': t('nav.sections.governance'),
   282→    'nav.sections.civicTools': t('nav.sections.civicTools'),
   283→    'nav.sections.citizenAction': t('nav.sections.citizenAction'),
   284→    'nav.sections.dataAlerts': t('nav.sections.dataAlerts'),
   285→    'nav.sections.analytics': t('nav.sections.analytics'),
   286→    'nav.sections.leadershipProjects': t('nav.sections.leadershipProjects'),
   287→    'nav.sections.insights': t('nav.sections.insights'),
   288→    'nav.sections.aiTools': t('nav.sections.aiTools'),
   289→  }), [t]);
   290→
   291→  // Helper to get localized label for a nav item
   292→  const getItemLabel = (id: TabId) => navItems.find(n => n.id === id)?.label ?? id;
   293→
   294→  // Helper to get localized section label
   295→  const getSectionLabel = (sectionKey: string) => sectionLabels[sectionKey] ?? sectionKey;
   296→
   297→  // Sidebar section renderer helper
   298→  const renderSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   299→    <div>
   300→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   301→      {items.map((item) => (
   302→        <button
   303→          key={item.id}
   304→          onClick={() => setActiveTab(item.id)}
   305→          aria-current={activeTab === item.id ? 'page' : undefined}
   306→          className={`w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-xs font-medium transition-colors mb-0.5 ${
   307→            activeTab === item.id
   308→              ? `${activeColor} text-white shadow-sm`
   309→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   310→          }`}
   311→        >
   312→          <item.icon className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   313→          <span>{getItemLabel(item.id)}</span>
   314→        </button>
   315→      ))}
   316→    </div>
   317→  );
   318→
   319→  // Mobile sidebar section renderer helper
   320→  const renderMobileSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   321→    <div>
   322→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   323→      {items.map((item) => (
   324→        <button
   325→          key={item.id}
   326→          onClick={() => { setActiveTab(item.id); setSidebarOpen(false); }}
   327→          aria-current={activeTab === item.id ? 'page' : undefined}
   328→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors mb-0.5 ${
   329→            activeTab === item.id
   330→              ? `${activeColor} text-white`
   331→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   332→          }`}
   333→        >
   334→          <item.icon className="h-4 w-4" aria-hidden="true" />
   335→          <span>{getItemLabel(item.id)}</span>
   336→        </button>
   337→      ))}
   338→    </div>
   339→  );
   340→
   341→  const { theme, setTheme } = useTheme();
   342→
   343→  // Keyboard shortcuts
   344→  React.useEffect(() => {
   345→    const handler = (e: KeyboardEvent) => {
   346→      if (e.key === 'Escape') { setSidebarOpen(false); return; }
   347→      if ((e.ctrlKey || e.metaKey) && e.key === 'k') { e.preventDefault(); commandPaletteRef.current?.open(); return; }
   348→      if ((e.ctrlKey || e.metaKey) && e.key === 'd') { e.preventDefault(); setTheme(theme === 'dark' ? 'light' : 'dark'); return; }
   349→      if (e.key === '?' && !['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) {
   350→        alert('Keyboard Shortcuts:\nCtrl+K: Search\nCtrl+D: Toggle Dark Mode\nEsc: Close panels\n?: This help');
   351→      }
   352→    };
   353→    window.addEventListener('keydown', handler);
   354→    return () => window.removeEventListener('keydown', handler);
   355→  }, [theme, setTheme]);
   356→
   357→  return (
   358→    <TooltipProvider>
   359→      <div className="min-h-screen flex flex-col bg-stone-50 dark:bg-stone-950">
   360→        {/* ══════════ SKIP NAVIGATION ══════════ */}
   361→        <a href="#main-content" className="skip-to-content">
   362→          Skip to main content
   363→        </a>
   364→
   365→        {/* ══════════ HEADER ══════════ */}
