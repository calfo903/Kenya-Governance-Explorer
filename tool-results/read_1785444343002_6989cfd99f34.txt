   146→// ─── Helper: interpolate color from red to green ────────────────────
   147→
   148→function lerpColor(t: number): string {
   149→  const clamped = Math.max(0, Math.min(1, t));
   150→  const r = Math.round(220 * (1 - clamped) + 22 * clamped);
   151→  const g = Math.round(38 * (1 - clamped) + 163 * clamped);
   152→  const b = Math.round(38 * (1 - clamped) + 74 * clamped);
   153→  return `rgb(${r}, ${g}, ${b})`;
   154→}

   156→// ─── Helper: population density color ───────────────────────────────
   157→
   158→function densityColor(pop: number, area: number): string {
   159→  const density = pop / area;
   160→  if (density > 2000) return "#7f1d1d";
   161→  if (density > 1000) return "#991b1b";
   162→  if (density > 500) return "#b91c1c";
   163→  if (density > 200) return "#c2410c";
   164→  if (density > 100) return "#ea580c";
   165→  if (density > 50) return "#f59e0b";
   166→  if (density > 20) return "#fbbf24";
   167→  return "#fef08a";
   168→}

   178→function buildAuditMap(fy?: string) {
   179→  const m: Record<string, (typeof countyAuditData)[number]> = {};
   180→  for (const r of countyAuditData) {
   181→    if (fy) {
   182→      // Filter to specific financial year
   183→      if (r.financialYear === fy) {
   184→        m[r.countyCode] = r;
   185→      }
   186→    } else {
   187→      // Pick latest
   188→      if (!m[r.countyCode] || r.financialYear > m[r.countyCode].financialYear) {
   189→        m[r.countyCode] = r;
   190→      }
   191→    }
   192→  }
   193→  return m;

   196→function buildBudgetAbsorptionMap(fy?: string) {
   197→  const m: Record<string, number> = {};
   198→  if (fy) {
   199→    // Filter to specific financial year, prefer full year records
   200→    for (const r of countyBudgetData) {
   201→      if (r.financialYear === fy && r.period === "Full Year") {
   202→        m[r.countyCode] = r.devAbsorptionRate;
   203→      }
   204→    }
   205→    // Fall back to any record in that FY if no full year record
   206→    for (const r of countyBudgetData) {
   207→      if (r.financialYear === fy && !m[r.countyCode]) {
   208→        m[r.countyCode] = r.devAbsorptionRate;
   209→      }
   210→    }
   211→  } else {
   212→    // Default: pick FY 2024/25 full year records
   213→    for (const r of countyBudgetData) {
   214→      if (!m[r.countyCode] || r.financialYear > "FY 2024/24") {
   215→        m[r.countyCode] = r.devAbsorptionRate;
   216→      }
   217→    }
   218→    // Ensure we pick the FY 2024/25 full year records
   219→    for (const r of countyBudgetData) {
   220→      if (r.financialYear === "FY 2024/25" && r.period === "Full Year") {
   221→        m[r.countyCode] = r.devAbsorptionRate;
   222→      }
   223→    }
   224→  }
   225→  return m;

   418→// ─── County color resolver ──────────────────────────────────────────
   419→
   420→function resolveCountyFill(
   421→  code: string,
   422→  mode: ColorMode,
   423→  govMap: Record<string, (typeof all47Governors)[number]>,
   424→  auditMap: Record<string, (typeof countyAuditData)[number]>,
   425→  budgetMap: Record<string, number>,
   426→  cecmScoreMap: Map<string, { overallScore: number }>,
   427→): string {
   428→  switch (mode) {
   429→    case "coalition": {
   430→      const gov = govMap[code];
   431→      return gov ? (COALITION_COLORS[gov.coalition] ?? DEFAULT_FILL) : DEFAULT_FILL;
   432→    }
   433→    case "region": {
   434→      const gov = govMap[code];
   435→      return gov ? (REGION_COLORS[gov.region] ?? DEFAULT_FILL) : DEFAULT_FILL;
   436→    }
   437→    case "audit": {
   438→      const audit = auditMap[code];
   439→      if (!audit || !audit.executiveOpinion) return DEFAULT_FILL;
   440→      return AUDIT_COLORS[audit.executiveOpinion] ?? DEFAULT_FILL;
   441→    }
   442→    case "budget": {
   443→      const rate = budgetMap[code];
   444→      if (rate == null) return DEFAULT_FILL;
   445→      return lerpColor(rate / 100);
   446→    }
   447→    case "cecm-performance": {
   448→      const scoreEntry = cecmScoreMap.get(code);
   449→      if (!scoreEntry) return DEFAULT_FILL;
   450→      return getCECMScoreColor(scoreEntry.overallScore);
   451→    }
   452→    case "population": {
   453→      const gov = govMap[code];
   454→      if (!gov) return DEFAULT_FILL;
   455→      return densityColor(gov.population, gov.areaSqKm);
   456→    }
   457→    default:
   458→      return DEFAULT_FILL;
   459→  }

   642→        onCountyClick?.(county.code, county.name);
   643→      },
   644→      [onCountyClick],
   645→    );
   646→
   647→    const handleKeyDown = useCallback(
   648→      (county: CountyShape, e: React.KeyboardEvent<SVGPathElement>) => {
   649→        if (e.key === 'Enter' || e.key === ' ') {
   650→          e.preventDefault();
   651→          handleClick(county);
   652→        }
   653→      },
   654→      [handleClick],
   655→    );
   656→
   657→    const handleMiniKeyDown = useCallback(
   658→      (county: CountyShape) => {
   659→        onCountyClick?.(county.code);
   660→      },
   661→      [onCountyClick],
   662→    );
   663→
   664→    return (
   665→      <div ref={containerRef} className={`relative w-full ${className}`}>
   666→        {/* Tooltip — positioned absolutely within outermost container (not affected by zoom) */}
   667→        {tooltipData && hoveredCounty && (
   668→          <MapTooltip data={tooltipData} x={tooltipPos.x} y={tooltipPos.y} />
   669→        )}
   670→
   671→        {/* Zoom/pan wrapper */}
   672→        <TransformWrapper
   673→          ref={transformRef}
   674→          initialScale={0.7}
   675→          minScale={0.5}
   676→          maxScale={5}
   677→          centerOnInit={true}
   678→        >
   679→          {({ zoomIn, zoomOut, resetView }) => (
   680→            <>
   681→              <TransformComponent
   682→                wrapperClass="!w-full !overflow-hidden"
   683→                contentClass="!w-full"
   684→              >
   685→                {/* SVG Map */}
   686→                <svg
   687→                  viewBox="0 0 1000 1200"
   688→                  className="w-full h-auto select-none"
   689→                  style={{ maxHeight: '600px' }}
   690→                  role="img"
   691→                  aria-label="Interactive map of Kenya showing 47 counties. Use arrow keys to navigate, Enter or Space to select a county."
   692→                >
   693→                  <defs>
   694→                    <filter id="county-shadow" x="-5%" y="-5%" width="110%" height="110%">
   695→                      <feDropShadow dx="0" dy="1" stdDeviation="1.5" floodOpacity="0.15" />
   696→                    </filter>
   697→                    <filter id="selected-glow" x="-10%" y="-10%" width="120%" height="120%">
   698→                      <feGaussianBlur stdDeviation="3" result="blur" />
   699→                      <feMerge>
   700→                        <feMergeNode in="blur" />
   701→                        <feMergeNode in="SourceGraphic" />
   702→                      </feMerge>
   703→                    </filter>
   704→                    <style>{svgStyle}</style>
   705→                  </defs>
   706→
   707→                  {/* Kenya outline border */}
   708→                  <path
   709→                    d={KENYA_OUTLINE}
   710→                    fill="none"
   711→                    stroke={isDark ? "#57534e" : "#1e293b"}
   712→                    strokeWidth="2.5"
   713→                    strokeLinejoin="round"
   714→                  />
   715→
   716→                  {/* County shapes */}
   717→                  {kenyaCountyPaths.map((county) => {
   718→                    const fill = resolveCountyFill(
   719→                      county.code,
   720→                      colorMode,
   721→                      govMap,
   722→                      auditMap,
   723→                      budgetMap,
   724→                      cecmScoreMap,
   725→                    );
   726→                    const isHovered = hoveredCounty === county.code;
   727→                    const isHighlighted = highlightedCounties.includes(county.code);
   728→                    const isSelected = selectedCounties.includes(county.code);
   729→                    const isSmall = SMALL_COUNTY_CODES.has(county.code);
   730→
   731→                    return (
   732→                      <g key={county.code}>
   733→                        <path
   734→                          d={county.path}
   735→                          fill={fill}
   736→                          className={[
   737→                            "county-path",
   738→                            isHighlighted ? "highlighted" : "",
   739→                            isSelected ? "selected county-selected-glow" : "",
   740→                            animated && isHighlighted ? "county-animated" : "",
   741→                          ].join(" ")}
   742→                          style={{
   743→                            opacity: isHovered ? 0.85 : 1,
   744→                          }}
   745→                          onMouseEnter={(e) => handleMouseEnter(county, e)}
   746→                          onMouseLeave={handleMouseLeave}
   747→                          onClick={() => handleClick(county)}
   748→                          onKeyDown={(e) => handleKeyDown(county, e)}
   749→                          tabIndex={0}
   750→                          role="button"
   751→                          aria-label={`${county.name} County`}
   752→                        />
   753→                        {showLabels && !isSmall && (
   754→                          <text
   755→                            x={county.cx}
   756→                            y={county.cy}
   757→                            className={isSmall ? "county-label-small" : "county-label"}
   758→                          >
   759→                            {county.name.length > 10
   760→                              ? county.name.split(" ").map((w, i) => (
   761→                                  <tspan
   762→                                    key={i}
   763→                                    x={county.cx}
   764→                                    dy={i === 0 ? "-0.4em" : "1.1em"}
   765→                                  >
   766→                                    {w}
   767→                                  </tspan>
   768→                                ))
   769→                              : county.name}
   770→                          </text>
   771→                        )}
   772→                      </g>
   773→                    );
   774→                  })}
   775→                </svg>
   776→              </TransformComponent>
   777→
   778→              {/* Zoom controls overlay — bottom-right */}
   779→              <div className="absolute bottom-4 right-4 z-10 flex flex-col gap-1">
   780→                <button
   781→                  type="button"
   782→                  onClick={() => zoomIn()}
   783→                  className="flex h-8 w-8 items-center justify-center rounded-md border border-border bg-background shadow-sm transition-colors hover:bg-accent"
   784→                  aria-label="Zoom in"
   785→                >
   786→                  <ZoomIn className="h-4 w-4" />
   787→                </button>
   788→                <button
   789→                  type="button"
   790→                  onClick={() => zoomOut()}
   791→                  className="flex h-8 w-8 items-center justify-center rounded-md border border-border bg-background shadow-sm transition-colors hover:bg-accent"
   792→                  aria-label="Zoom out"
   793→                >
   794→                  <ZoomOut className="h-4 w-4" />
   795→                </button>
   796→                <button
   797→                  type="button"
   798→                  onClick={() => resetView()}
   799→                  className="flex h-8 w-8 items-center justify-center rounded-md border border-border bg-background shadow-sm transition-colors hover:bg-accent"
   800→                  aria-label="Reset zoom"
   801→                >
   802→                  <Maximize2 className="h-4 w-4" />
   803→                </button>
   804→              </div>
   805→            </>
   806→          )}
   807→        </TransformWrapper>
   808→
   809→        {/* Legend */}
   810→        <div className="mt-3 flex flex-wrap items-center justify-center gap-x-4 gap-y-2 px-2">
   811→          <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   812→            <Info className="h-3 w-3" />
   813→            <span className="font-medium capitalize">
   814→              {colorMode === "coalition"
   815→                ? "Coalition"
   816→                : colorMode === "region"
   817→                  ? "Region"
   818→                  : colorMode === "audit"
   819→                    ? "Audit Opinion"
   820→                    : colorMode === "budget"
   821→                      ? "Budget Absorption"
   822→                      : colorMode === "cecm-performance"
   823→                        ? "CECM Performance"
   824→                        : "Population Density"}
   825→            </span>
   826→          </div>
   827→          {legendItems.map((item, i) => (
   828→            <div key={i} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   829→              <span
   830→                className="inline-block h-3 w-3 rounded-sm border border-white/30 dark:border-stone-600 shadow-sm"
   831→                style={{ backgroundColor: item.color }}
   832→              />
   833→              <span>{item.label}</span>
   834→            </div>
   835→          ))}
   836→        </div>
   837→      </div>
   838→    );
   839→  },
