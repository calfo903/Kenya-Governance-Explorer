   198→function PageContent() {
   199→  const t = useTranslations();
   200→  const locale = useLocaleStore((s) => s.locale);
   201→  const [activeTab, setActiveTab] = useState<TabId>('summary');
   202→  const [selectedCounty, setSelectedCounty] = useState<string>('034');
   203→  const [expandedCounties, setExpandedCounties] = useState<Set<string>>(new Set());
   204→  const [comparisonList, setComparisonList] = useState<ComparisonItem[]>([]);
   205→  const [filters, setFilters] = useState<FilterState>({});
   206→  const [sidebarOpen, setSidebarOpen] = useState(false);
   207→  const commandPaletteRef = React.useRef<{ open: () => void; close: () => void }>(null);
   208→
   209→  // Localized nav items
   210→  const navItems = useMemo(() => navItemDefs.map(def => ({
   211→    ...def,
   212→    label: t(def.labelKey),
   213→  })), [t]);
   214→
   215→  const allCounties = useMemo((): County[] => {
   216→    const placeholders = getPlaceholderCounties();
   217→    return placeholders.map((c) => {
   218→      if (c.code === '034') return kajiadoCounty;
   219→      // Inject real audit opinion from county-audit-data
   220→      const auditRecords = getCountyAuditRecords(c.code);
   221→      const latestAudit = auditRecords.find(a => a.financialYear === 'FY 2024/25');
   222→      const latestBudget = getCountyBudget(c.code, 'FY 2024/25');
   223→      return {
   224→        ...c,
   225→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   226→        executiveAuditSource: latestAudit?.source,
   227→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
   228→        developmentAbsorptionSource: latestBudget ? {
   229→          source: latestBudget.source.source,
   230→          reportTitle: latestBudget.source.reportTitle,
   231→          financialYear: latestBudget.source.financialYear,
   232→          url: latestBudget.source.url,
   233→          accessedDate: latestBudget.source.accessedDate,
   234→        } : undefined,
   235→        dataAvailability: 'partial' as const,
   236→        dataAvailabilityNote: `Audit opinion: ${latestAudit?.executiveOpinion || 'pending'} (FY 2024/25). Budget absorption: ${latestBudget?.devAbsorptionRate ?? 'pending'}%. Full official data from oagkenya.go.ke and cob.go.ke.`,
   237→      };
   238→    });
   239→  }, []);
   240→
   241→  const toggleCounty = (code: string) => {
   242→    const next = new Set(expandedCounties);
   243→    if (next.has(code)) next.delete(code); else next.add(code);
   244→    setExpandedCounties(next);
   245→  };
   246→
   247→  const addToComparison = (rep: Representative, countyName: string) => {
   248→    if (comparisonList.length >= 4) return;
   249→    if (comparisonList.find(c => c.representative.id === rep.id)) return;
   250→    setComparisonList([...comparisonList, { representative: rep, countyName }]);
   251→  };
   252→
   253→  const removeFromComparison = (id: string) => {
   254→    setComparisonList(comparisonList.filter(c => c.representative.id !== id));
   255→  };
   256→
   257→  const filteredGovernors = useMemo(() => {
   258→    return all47Governors.filter((g) => {
   259→      if (filters.region && g.region !== filters.region) return false;
   260→      if (filters.coalition && g.coalition !== filters.coalition) return false;
   261→      if (filters.party && g.party !== filters.party) return false;
   262→      if (filters.keyword) {
   263→        const kw = filters.keyword.toLowerCase();
   264→        if (!g.name.toLowerCase().includes(kw) && !g.county.toLowerCase().includes(kw)) return false;
   265→      }
   266→      return true;
   267→    });
   268→  }, [filters]);
   269→
   270→  // Group nav items by section
   271→  const governanceItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.governance');
   272→  const civicItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.civicTools');
   273→  const citizenItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.citizenAction');
   274→  const dataItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.dataAlerts');
   275→  const analyticsItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.analytics');
   276→  const leadershipItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.leadershipProjects');
   277→  const insightItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.insights');
   278→  const aiItems = navItemDefs.filter(n => n.sectionKey === 'nav.sections.aiTools');
   279→
   280→  // Section labels lookup
   281→  const sectionLabels = useMemo(() => ({
   282→    'nav.sections.governance': t('nav.sections.governance'),
   283→    'nav.sections.civicTools': t('nav.sections.civicTools'),
   284→    'nav.sections.citizenAction': t('nav.sections.citizenAction'),
   285→    'nav.sections.dataAlerts': t('nav.sections.dataAlerts'),
   286→    'nav.sections.analytics': t('nav.sections.analytics'),
   287→    'nav.sections.leadershipProjects': t('nav.sections.leadershipProjects'),
   288→    'nav.sections.insights': t('nav.sections.insights'),
   289→    'nav.sections.aiTools': t('nav.sections.aiTools'),
   290→  }), [t]);
   291→
   292→  // Helper to get localized label for a nav item
   293→  const getItemLabel = (id: TabId) => navItems.find(n => n.id === id)?.label ?? id;
   294→
   295→  // Helper to get localized section label
   296→  const getSectionLabel = (sectionKey: string) => sectionLabels[sectionKey] ?? sectionKey;
   297→
   298→  // Sidebar section renderer helper
   299→  const renderSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   300→    <div>
   301→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   302→      {items.map((item) => (
   303→        <button
   304→          key={item.id}
   305→          onClick={() => setActiveTab(item.id)}
   306→          aria-current={activeTab === item.id ? 'page' : undefined}
   307→          className={`w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-xs font-medium transition-colors mb-0.5 ${
   308→            activeTab === item.id
   309→              ? `${activeColor} text-white shadow-sm`
   310→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   311→          }`}
   312→        >
   313→          <item.icon className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   314→          <span>{getItemLabel(item.id)}</span>
   315→        </button>
   316→      ))}
   317→    </div>
   318→  );
   319→
   320→  // Mobile sidebar section renderer helper
   321→  const renderMobileSidebarSection = (items: NavItem[], sectionKey: string, activeColor: string) => (
   322→    <div>
   323→      <p className="px-3 py-1.5 text-[10px] font-semibold text-stone-500 dark:text-stone-400 uppercase tracking-wider">{getSectionLabel(sectionKey)}</p>
   324→      {items.map((item) => (
   325→        <button
   326→          key={item.id}
   327→          onClick={() => { setActiveTab(item.id); setSidebarOpen(false); }}
   328→          aria-current={activeTab === item.id ? 'page' : undefined}
   329→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors mb-0.5 ${
   330→            activeTab === item.id
   331→              ? `${activeColor} text-white`
   332→              : 'text-stone-600 dark:text-stone-300 hover:bg-stone-50 dark:hover:bg-stone-800'
   333→          }`}
   334→        >
   335→          <item.icon className="h-4 w-4" aria-hidden="true" />
   336→          <span>{getItemLabel(item.id)}</span>
   337→        </button>
   338→      ))}
   339→    </div>
   340→  );
   341→
   342→  const { theme, setTheme } = useTheme();
   343→
   344→  // Keyboard shortcuts
   345→  React.useEffect(() => {
   346→    const handler = (e: KeyboardEvent) => {
   347→      if (e.key === 'Escape') { setSidebarOpen(false); return; }
   348→      if ((e.ctrlKey || e.metaKey) && e.key === 'k') { e.preventDefault(); commandPaletteRef.current?.open(); return; }
   349→      if ((e.ctrlKey || e.metaKey) && e.key === 'd') { e.preventDefault(); setTheme(theme === 'dark' ? 'light' : 'dark'); return; }
   350→      if (e.key === '?' && !['INPUT', 'TEXTAREA', 'SELECT'].includes((e.target as HTMLElement).tagName)) {
   351→        alert('Keyboard Shortcuts:\nCtrl+K: Search\nCtrl+D: Toggle Dark Mode\nEsc: Close panels\n?: This help');
   352→      }
   353→    };
   354→    window.addEventListener('keydown', handler);
   355→    return () => window.removeEventListener('keydown', handler);
   356→  }, [theme, setTheme]);
   357→
   358→  return (
   359→    <TooltipProvider>
   360→      <div className="min-h-screen flex flex-col bg-stone-50 dark:bg-stone-950">
   361→        {/* ══════════ SKIP NAVIGATION ══════════ */}
   362→        <a href="#main-content" className="skip-to-content">
   363→          Skip to main content
   364→        </a>
   365→
