   239→  'bg-emerald-100 text-emerald-700',

   242→  'bg-sky-100 text-sky-700',

   249→function getAvatarColor(name: string): string {
   250→  let hash = 0;
   251→  for (let i = 0; i < name.length; i++) {
   252→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   253→  }
   254→  return AVATAR_COLORS[Math.abs(hash) % AVATAR_COLORS.length];
   255→}

   257→function portfolioIcon(portfolio: string) {
   258→  const lower = portfolio.toLowerCase();
   259→  if (lower.includes('health') || lower.includes('medical')) return <Stethoscope className="h-3.5 w-3.5 text-emerald-600" />;
   260→  if (lower.includes('education') || lower.includes('learn')) return <GraduationCap className="h-3.5 w-3.5 text-emerald-600" />;
   261→  if (lower.includes('water') || lower.includes('irrigation')) return <Droplets className="h-3.5 w-3.5 text-emerald-600" />;
   262→  if (lower.includes('road') || lower.includes('infra') || lower.includes('transport')) return <Truck className="h-3.5 w-3.5 text-emerald-600" />;
   263→  if (lower.includes('agric') || lower.includes('livestock')) return <Leaf className="h-3.5 w-3.5 text-emerald-600" />;
   264→  if (lower.includes('finance') || lower.includes('treasury') || lower.includes('economic')) return <DollarSign className="h-3.5 w-3.5 text-emerald-600" />;
   265→  if (lower.includes('trade') || lower.includes('cooper')) return <Handshake className="h-3.5 w-3.5 text-emerald-600" />;
   266→  if (lower.includes('land') || lower.includes('housing') || lower.includes('urban')) return <Building2 className="h-3.5 w-3.5 text-emerald-600" />;
   267→  if (lower.includes('youth') || lower.includes('gender') || lower.includes('sport') || lower.includes('culture')) return <Award className="h-3.5 w-3.5 text-emerald-600" />;
   268→  if (lower.includes('admin') || lower.includes('devolution') || lower.includes('service')) return <Briefcase className="h-3.5 w-3.5 text-emerald-600" />;
   269→  if (lower.includes('tourism') || lower.includes('environment')) return <MapPin className="h-3.5 w-3.5 text-emerald-600" />;

   284→function TreeNode({
   285→  label,
   286→  icon,
   287→  children,
   288→  defaultOpen = false,
   289→  badge,
   290→  forceOpen,
   291→}: {
   292→  label: string;
   293→  icon: React.ReactNode;
   294→  children: React.ReactNode;
   295→  defaultOpen?: boolean;
   296→  badge?: string;
   297→  forceOpen?: boolean | null;
   298→}) {
   299→  const [internalOpen, setInternalOpen] = useState(defaultOpen);
   300→  const open = forceOpen !== undefined && forceOpen !== null ? forceOpen : internalOpen;
   301→  return (
   302→    <div>
   303→      <button
   304→        onClick={() => {
   305→          if (forceOpen === undefined || forceOpen === null) setInternalOpen(!internalOpen);
   306→        }}
   307→        className="flex items-center gap-2 w-full text-left py-1.5 px-1 rounded-md hover:bg-emerald-50 dark:bg-emerald-950 transition-colors group"
   308→      >
   309→        {open ? (
   310→          <ChevronDown className="h-3 w-3 text-emerald-600 shrink-0" />
   311→        ) : (
   312→          <ChevronRight className="h-3 w-3 text-emerald-600 shrink-0" />
   313→        )}
   314→        <span className="shrink-0">{icon}</span>
   315→        <span className="text-xs font-semibold text-gray-800 dark:text-gray-100">{label}</span>
   316→        {badge && (
   317→          <span className="text-[10px] text-emerald-600 font-medium bg-emerald-50 dark:bg-emerald-950 px-1.5 py-0.5 rounded-full">
   318→            {badge}

   320→        )}
   321→      </button>
   322→      {open && (
   323→        <div className="ml-4 border-l-2 border-emerald-200 pl-3 mt-1 space-y-1">{children}</div>
   324→      )}
   325→    </div>

   421→function AuditBadge({ opinion }: { opinion: AuditOpinion | string | null | undefined }) {

   432→    Adverse: 'bg-orange-100 text-orange-700 border-orange-300',
   433→    Disclaimer: 'bg-red-100 text-red-700 border-red-300',
   434→  };
   435→  const colorClass = colorMap[opinion] || 'bg-gray-100 dark:bg-gray-800 text-gray-500 dark:text-gray-400 border-gray-300 dark:border-gray-600';
   436→  return (
   437→    <span className={`text-[10px] px-2 py-0.5 rounded-full border font-medium ${colorClass}`}>
   438→      {opinion}
   439→    </span>
   440→  );
