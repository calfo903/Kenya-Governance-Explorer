    41→function getAttendanceColor(rate: number): string {
    42→  if (rate > 85) return 'text-emerald-600';
    43→  if (rate >= 70) return 'text-amber-600';
    44→  return 'text-red-600';
    45→}

    47→function getAttendanceBarColor(rate: number): string {
    48→  if (rate > 85) return '[&>div]:!bg-emerald-500';
    49→  if (rate >= 70) return '[&>div]:!bg-amber-500';
    50→  return '[&>div]:!bg-red-500';

    53→function getAttendanceBadgeClass(rate: number): string {

    56→  return 'bg-red-100 text-red-700 border-red-200';

   327→// ─── Statistics Panel ───────────────────────────────────────────
   328→function StatisticsPanel({ members }: { members: MzalendoMember[] }) {
   329→  const totalMembers = members.length;
   330→  const maleCount = members.filter(m => m.gender === 'male').length;
   331→  const femaleCount = members.filter(m => m.gender === 'female').length;
   332→
   333→  const coalitions = useMemo(() => {
   334→    const counts: Record<string, number> = {};
   335→    members.forEach(m => {
   336→      const c = m.coalition || 'Unknown';
   337→      counts[c] = (counts[c] || 0) + 1;
   338→    });
   339→    return counts;
   340→  }, [members]);
   341→
   342→  const membersWithAttendance = members.filter(m => m.voteRecord);
   343→  const avgAttendance = membersWithAttendance.length > 0
   344→    ? Math.round(membersWithAttendance.reduce((sum, m) => sum + (m.voteRecord?.attendanceRate || 0), 0) / membersWithAttendance.length)
   345→    : 0;
   346→
   347→  return (
   348→    <Card className="border-stone-200 dark:border-stone-700 dark:bg-stone-900">
   349→      <CardHeader className="pb-3">
   350→        <CardTitle className="text-sm font-bold flex items-center gap-2 text-stone-900 dark:text-stone-50 dark:text-stone-100">
   351→          <BarChart3 className="h-4 w-4 text-emerald-600" />
   352→          Overview Statistics
   353→        </CardTitle>
   354→        <CardDescription className="text-xs text-stone-500 dark:text-stone-400">
   355→          Based on {totalMembers} members in the current dataset
   356→        </CardDescription>
   357→      </CardHeader>
   358→      <CardContent>
   359→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   360→          {/* Total */}
   361→          <div className="text-center space-y-1">
   362→            <div className="flex items-center justify-center gap-1.5">
   363→              <Users className="h-4 w-4 text-stone-400" />

   369→          {/* Gender ratio */}
   370→          <div className="text-center space-y-1">
   371→            <div className="flex items-center justify-center gap-2">
   372→              <span className="text-lg font-bold text-sky-600 dark:text-sky-400">{maleCount}</span>
   373→              <span className="text-stone-300 dark:text-stone-600">/</span>
   374→              <span className="text-lg font-bold text-pink-600 dark:text-pink-400">{femaleCount}</span>
   375→            </div>
   376→            <p className="text-xs text-stone-500 dark:text-stone-400">Male / Female</p>
   377→          </div>
   378→
   379→          {/* Coalition distribution */}
   380→          <div className="text-center space-y-1">
   381→            <div className="flex items-center justify-center gap-1 flex-wrap">
   382→              {Object.entries(coalitions).map(([name, count]) => {
   383→                const label = name === 'Kenya Kwanza Alliance'
   384→                  ? 'KK'
   385→                  : name === 'Azimio la Umoja One Kenya Coalition'
   386→                    ? 'AZ'
   387→                    : 'IND';
   388→                const color = COALITION_DOT_COLORS[name] || 'bg-gray-500';
   389→                return (
   390→                  <span key={name} className="flex items-center gap-0.5 text-xs">
   391→                    <span className={`h-2 w-2 rounded-full ${color}`} />
   392→                    <span className="font-semibold text-stone-700 dark:text-stone-200 dark:text-stone-300">{count}</span>
   393→                    <span className="text-stone-400 dark:text-stone-500">{label}</span>
   394→                  </span>
   395→                );
   396→              })}
   397→            </div>
   398→            <p className="text-xs text-stone-500 dark:text-stone-400">Coalitions</p>
   399→          </div>
   400→
   401→          {/* Average attendance */}
   402→          <div className="text-center space-y-1">
   403→            <span className={`text-2xl font-bold ${getAttendanceColor(avgAttendance)}`}>
   404→              {avgAttendance}%

   406→            <p className="text-xs text-stone-500 dark:text-stone-400">Avg Attendance</p>
   407→          </div>
   408→        </div>
   409→
   410→        {/* Gender visual bar */}
   411→        <div className="mt-4">
   412→          <div className="flex h-2 rounded-full overflow-hidden">
   413→            {totalMembers > 0 && (
   414→              <>
   415→                <div
   416→                  className="bg-sky-400 dark:bg-sky-500 transition-all"
   417→                  style={{ width: `${(maleCount / totalMembers) * 100}%` }}
   418→                />
   419→                <div
   420→                  className="bg-pink-400 dark:bg-pink-500 transition-all"
   421→                  style={{ width: `${(femaleCount / totalMembers) * 100}%` }}

   424→            )}
   425→          </div>
   426→          <div className="flex items-center justify-between mt-1.5">
   427→            <span className="text-xs text-stone-400 dark:text-stone-500 flex items-center gap-1">
