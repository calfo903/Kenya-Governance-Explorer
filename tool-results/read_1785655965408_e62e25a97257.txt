    33→function getSeverityColor(severity: EventSeverity | undefined): { bg: string; border: string; text: string; dot: string } {
    34→  switch (severity) {
    35→    case 'critical':
    36→      return { bg: 'bg-red-100', border: 'border-red-300', text: 'text-red-800', dot: 'bg-red-500' };
    37→    case 'warning':
    38→      return { bg: 'bg-amber-100', border: 'border-amber-300', text: 'text-amber-800', dot: 'bg-amber-500' };
    39→    case 'success':
    40→      return { bg: 'bg-emerald-100', border: 'border-emerald-300', text: 'text-emerald-800', dot: 'bg-emerald-500' };
    41→    case 'info':
    42→    default:
    43→      return { bg: 'bg-blue-100', border: 'border-blue-300', text: 'text-blue-800', dot: 'bg-blue-500' };
    44→  }
    45→}

    84→function formatCurrency(amount: number): string {
    85→  if (amount >= 1_000_000_000) return `KES ${(amount / 1_000_000_000).toFixed(1)}B`;
    86→  if (amount >= 1_000_000) return `KES ${(amount / 1_000_000).toFixed(1)}M`;
    87→  if (amount >= 1_000) return `KES ${(amount / 1_000).toFixed(1)}K`;
    88→  return `KES ${amount.toLocaleString()}`;
    89→}

    91→function formatDate(dateStr: string): string {

    93→    const d = new Date(dateStr);
    94→    return d.toLocaleDateString('en-KE', { year: 'numeric', month: 'short', day: 'numeric' });
    95→  } catch {
    96→    return dateStr;
    97→  }
    98→}

   100→function getRiskLevel(score: number | undefined): { label: string; color: string; textColor: string } {
   101→  if (score === undefined) return { label: 'N/A', color: 'bg-stone-200', textColor: 'text-stone-600 dark:text-stone-300' };
   102→  if (score >= 75) return { label: 'High', color: 'bg-red-500', textColor: 'text-red-600' };
   103→  if (score >= 50) return { label: 'Medium', color: 'bg-amber-500', textColor: 'text-amber-600' };
   104→  return { label: 'Low', color: 'bg-emerald-500', textColor: 'text-emerald-600' };
   105→}

   109→function RiskScoreGauge({ score }: { score: number | undefined }) {
   110→  const risk = getRiskLevel(score);
   111→  const displayScore = score !== undefined ? score : 0;
   112→
   113→  const circumference = 2 * Math.PI * 40;
   114→  const strokeDashoffset = circumference - (displayScore / 100) * circumference;
   115→
   116→  return (
   117→    <div className="flex flex-col items-center gap-1">
   118→      <div className="relative w-24 h-24">
   119→        <svg className="w-24 h-24 -rotate-90" viewBox="0 0 100 100">
   120→          <circle
   121→            cx="50"
   122→            cy="50"
   123→            r="40"
   124→            fill="none"
   125→            stroke="currentColor"
   126→            strokeWidth="8"
   127→            className="text-stone-200"
   128→          />
   129→          <circle
   130→            cx="50"
   131→            cy="50"
   132→            r="40"
   133→            fill="none"
   134→            stroke="currentColor"
   135→            strokeWidth="8"
   136→            strokeLinecap="round"
   137→            className={risk.color}
   138→            strokeDasharray={circumference}
   139→            strokeDashoffset={strokeDashoffset}
   140→            style={{ transition: 'stroke-dashoffset 0.8s ease-out' }}
   141→          />
   142→        </svg>
   143→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   144→          <span className={`text-xl font-bold ${risk.textColor}`}>
   145→            {score !== undefined ? score : '—'}

   148→        </div>
   149→      </div>
   150→      <span className={`text-xs font-semibold ${risk.textColor}`}>{risk.label} Risk</span>
   151→    </div>
   152→  );
   153→}

   274→function ProjectVelocityChart({ timeline }: { timeline: AuditTimelineEvent[] }) {
   275→  const chartData = useMemo(() => {
   276→    const monthMap = new Map<string, number>();
   277→    const now = new Date();
   278→
   279→    for (let i = 5; i >= 0; i--) {
   280→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   281→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   282→      monthMap.set(key, 0);
   283→    }
   284→
   285→    timeline.forEach((event) => {
   286→      const eventDate = new Date(event.date);
   287→      const key = eventDate.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   288→      if (monthMap.has(key)) {
   289→        monthMap.set(key, (monthMap.get(key) || 0) + 1);
   290→      }
   291→    });
   292→
   293→    return Array.from(monthMap.entries()).map(([month, count]) => ({
   294→      month,
   295→      events: count,
   296→    }));
   297→  }, [timeline]);
   298→
   299→  const maxEvents = Math.max(...chartData.map((d) => d.events), 1);
   300→
   301→  return (
   302→    <Card className="border-stone-200 dark:border-stone-700 bg-white dark:bg-stone-900">
   303→      <CardHeader className="pb-2 pt-4 px-4">
   304→        <CardTitle className="text-xs font-semibold text-stone-700 dark:text-stone-200 flex items-center gap-1.5">
   305→          <TrendingUp className="h-3.5 w-3.5 text-emerald-600" />
   306→          Audit Velocity
   307→        </CardTitle>
   308→        <CardDescription className="text-[10px]">Events per month (last 6 months)</CardDescription>
   309→      </CardHeader>
   310→      <CardContent className="px-4 pb-4">
   311→        <div className="h-28">
   312→          <ResponsiveContainer width="100%" height="100%">
   313→            <LineChart data={chartData}>
   314→              <CartesianGrid strokeDasharray="3 3" stroke="#e7e5e4" />
   315→              <XAxis
   316→                dataKey="month"
   317→                tick={{ fontSize: 10, fill: '#78716c' }}
   318→                axisLine={{ stroke: '#d6d3d1' }}
   319→                tickLine={false}
   320→              />
   321→              <YAxis
   322→                domain={[0, Math.max(maxEvents + 1, 4)]}
   323→                tick={{ fontSize: 10, fill: '#78716c' }}
   324→                axisLine={false}
   325→                tickLine={false}
   326→                width={24}
   327→              />
   328→              <RechartsTooltip
   329→                contentStyle={{
   330→                  backgroundColor: '#fff',
   331→                  border: '1px solid #d6d3d1',
   332→                  borderRadius: '6px',
   333→                  fontSize: '11px',
   334→                  padding: '4px 8px',
   335→                }}
   336→                formatter={(value: number) => [`${value} event${value !== 1 ? 's' : ''}`, 'Count']}
   337→              />
   338→              <Line
   339→                type="monotone"
   340→                dataKey="events"
   341→                stroke="#10b981"
   342→                strokeWidth={2}
   343→                dot={{ fill: '#10b981', r: 3 }}
   344→                activeDot={{ r: 5, stroke: '#10b981', strokeWidth: 2, fill: '#fff' }}
   345→              />
   346→            </LineChart>
   347→          </ResponsiveContainer>
   348→        </div>
   349→      </CardContent>
   350→    </Card>
   351→  );
   352→}

   391→function RiskForecastPanel({ project }: { project: ProjectRecord }) {
   392→  const forecast = useMemo((): RiskForecast => {
   393→    const riskScore = project.riskScore ?? 30;
   394→    const riskFactorCount = project.riskFactors?.length ?? 0;
   395→    const stallingProbability = Math.min(
   396→      95,
   397→      Math.max(5, Math.round((riskScore * 0.6) + (riskFactorCount * 8)))
   398→    );
   399→    const budgetVelocity = project.budgetAllocated > 0
   400→      ? Math.round((project.budgetSpent / project.budgetAllocated) * 100)
   401→      : 0;
   402→
   403→    return {
   404→      projectId: project.id,
   405→      stallingProbability,
   406→      factors: [
   407→        {
   408→          name: 'Budget Absorption',
   409→          weight: budgetVelocity > 80 ? 0.3 : budgetVelocity > 50 ? 0.15 : 0.35,
   410→          status: budgetVelocity > 80 ? 'normal' : budgetVelocity > 50 ? 'concerning' : 'critical',
   411→        },
   412→        {
   413→          name: 'Risk Score',
   414→          weight: riskScore > 60 ? 0.35 : riskScore > 30 ? 0.2 : 0.1,
   415→          status: riskScore > 60 ? 'critical' : riskScore > 30 ? 'concerning' : 'normal',
   416→        },
   417→        {
   418→          name: 'Risk Factor Count',
   419→          weight: Math.min(riskFactorCount * 0.1, 0.3),
   420→          status: riskFactorCount > 3 ? 'critical' : riskFactorCount > 1 ? 'concerning' : 'normal',
   421→        },
   422→      ],
   423→      predictedCompletionDate: project.endDate,
   424→      budgetVelocity,
   425→      milestoneCompletionRate: project.status === 'completed' ? 100 : project.status === 'active' ? 65 : 30,
   426→      recommendation:
   427→        stallingProbability > 70
   428→          ? 'Immediate intervention recommended. Escalate to OAG for priority review.'
   429→          : stallingProbability > 40
   430→          ? 'Monitor closely. Budget absorption below target — engage county assembly oversight.'
   431→          : 'Project on track. Continue routine monitoring and citizen auditing.',
   432→    };
   433→  }, [project]);
   434→
   435→  const stallingColor =
   436→    forecast.stallingProbability > 70
   437→      ? 'text-red-600'
   438→      : forecast.stallingProbability > 40
   439→      ? 'text-amber-600'
   440→      : 'text-emerald-600';
   441→
   442→  const stallingBarColor =
   443→    forecast.stallingProbability > 70
   444→      ? 'bg-red-500'
   445→      : forecast.stallingProbability > 40
   446→      ? 'bg-amber-500'
   447→      : 'bg-emerald-500';
   448→
   449→  return (
   450→    <Card className="border-stone-200 dark:border-stone-700 bg-white dark:bg-stone-900">
   451→      <CardHeader className="pb-2 pt-4 px-4">
   452→        <CardTitle className="text-xs font-semibold text-stone-700 dark:text-stone-200 flex items-center gap-1.5">
   453→          <Gauge className="h-3.5 w-3.5 text-amber-600" />
   454→          Risk Forecast
   455→        </CardTitle>
   456→        <CardDescription className="text-[10px]">Stalling probability analysis</CardDescription>
   457→      </CardHeader>
   458→      <CardContent className="px-4 pb-4 space-y-3">
   459→        {/* Stalling probability */}
   460→        <div className="flex items-center gap-2">
   461→          <div className="flex-1">
   462→            <div className="flex justify-between text-[11px] mb-1">
   463→              <span className="text-stone-600 dark:text-stone-300">Stalling Probability</span>
   464→              <span className={`font-bold ${stallingColor}`}>
   465→                {forecast.stallingProbability}%
   466→              </span>
   467→            </div>
   468→            <div className="h-1.5 rounded-full bg-stone-100 dark:bg-stone-700 overflow-hidden">
   469→              <div
   470→                className={`h-full rounded-full transition-all duration-700 ${stallingBarColor}`}
   471→                style={{ width: `${forecast.stallingProbability}%` }}
   472→              />
   473→            </div>
   474→          </div>
   475→        </div>
   476→
   477→        {/* Factors */}
   478→        <div className="space-y-1.5">
   479→          {forecast.factors.map((factor) => (
   480→            <div key={factor.name} className="flex items-center justify-between text-[11px]">
   481→              <span className="text-stone-500 dark:text-stone-400">{factor.name}</span>
   482→              <Badge
   483→                variant="outline"
   484→                className={`text-[9px] px-1.5 py-0 h-4 border ${
   485→                  factor.status === 'critical'
   486→                    ? 'bg-red-50 dark:bg-red-950 text-red-600 border-red-200'
