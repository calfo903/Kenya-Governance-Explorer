   259→        ...c,
   260→        executiveAuditOpinion: latestAudit?.executiveOpinion || undefined,
   261→        executiveAuditSource: latestAudit?.source,
   262→        developmentAbsorptionRate: latestBudget?.devAbsorptionRate || undefined,
