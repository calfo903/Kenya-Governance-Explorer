    34→function getSeverityColor(severity: EventSeverity | undefined): { bg: string; border: string; text: string; dot: string } {
    35→  switch (severity) {
    36→    case 'critical':
    37→      return { bg: 'bg-red-100', border: 'border-red-300', text: 'text-red-800', dot: 'bg-red-500' };
    38→    case 'warning':
    39→      return { bg: 'bg-amber-100', border: 'border-amber-300', text: 'text-amber-800', dot: 'bg-amber-500' };
    40→    case 'success':
    41→      return { bg: 'bg-emerald-100', border: 'border-emerald-300', text: 'text-emerald-800', dot: 'bg-emerald-500' };
    42→    case 'info':
    43→    default:
    44→      return { bg: 'bg-blue-100', border: 'border-blue-300', text: 'text-blue-800', dot: 'bg-blue-500' };
    45→  }
    46→}

    85→function formatCurrency(amount: number): string {
    86→  if (amount >= 1_000_000_000) return `KES ${(amount / 1_000_000_000).toFixed(1)}B`;
    87→  if (amount >= 1_000_000) return `KES ${(amount / 1_000_000).toFixed(1)}M`;
    88→  if (amount >= 1_000) return `KES ${(amount / 1_000).toFixed(1)}K`;
    89→  return `KES ${amount.toLocaleString()}`;
    90→}

    92→function formatDate(dateStr: string): string {

    94→    const d = new Date(dateStr);
    95→    return d.toLocaleDateString('en-KE', { year: 'numeric', month: 'short', day: 'numeric' });
    96→  } catch {
    97→    return dateStr;
    98→  }
    99→}

   101→function getRiskLevel(score: number | undefined): { label: string; color: string; textColor: string } {
   102→  if (score === undefined) return { label: 'N/A', color: 'bg-stone-200', textColor: 'text-stone-600 dark:text-stone-300' };
   103→  if (score >= 75) return { label: 'High', color: 'bg-red-500', textColor: 'text-red-600' };
   104→  if (score >= 50) return { label: 'Medium', color: 'bg-amber-500', textColor: 'text-amber-600' };
   105→  return { label: 'Low', color: 'bg-emerald-500', textColor: 'text-emerald-600' };
   106→}

   110→function RiskScoreGauge({ score }: { score: number | undefined }) {

   112→  const displayScore = score !== undefined ? score : 0;
   113→

   116→
   117→  return (
   118→    <div className="flex flex-col items-center gap-1">
   119→      <div className="relative w-24 h-24">
   120→        <svg className="w-24 h-24 -rotate-90" viewBox="0 0 100 100">
   121→          <circle
   122→            cx="50"
   123→            cy="50"
   124→            r="40"
   125→            fill="none"
   126→            stroke="currentColor"
   127→            strokeWidth="8"
   128→            className="text-stone-200"
   129→          />
   130→          <circle
   131→            cx="50"
   132→            cy="50"
   133→            r="40"
   134→            fill="none"
   135→            stroke="currentColor"
   136→            strokeWidth="8"
   137→            strokeLinecap="round"
   138→            className={risk.color}
   139→            strokeDasharray={circumference}
   140→            strokeDashoffset={strokeDashoffset}
   141→            style={{ transition: 'stroke-dashoffset 0.8s ease-out' }}
   142→          />
   143→        </svg>
   144→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   145→          <span className={`text-xl font-bold ${risk.textColor}`}>
   146→            {score !== undefined ? score : '—'}

   150→      </div>

   153→  );
   154→}

   275→function ProjectVelocityChart({ timeline }: { timeline: AuditTimelineEvent[] }) {
   276→  const chartData = useMemo(() => {
   277→    const monthMap = new Map<string, number>();
   278→    const now = new Date();
   279→
   280→    for (let i = 5; i >= 0; i--) {
   281→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   282→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   283→      monthMap.set(key, 0);
   284→    }
   285→
   286→    timeline.forEach((event) => {
   287→      const eventDate = new Date(event.date);
   288→      const key = eventDate.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   289→      if (monthMap.has(key)) {
   290→        monthMap.set(key, (monthMap.get(key) || 0) + 1);
   291→      }
   292→    });
   293→
   294→    return Array.from(monthMap.entries()).map(([month, count]) => ({
   295→      month,
   296→      events: count,
   297→    }));
   298→  }, [timeline]);
   299→
   300→  const maxEvents = Math.max(...chartData.map((d) => d.events), 1);
   301→
   302→  return (
   303→    <Card className="border-stone-200 dark:border-stone-700 bg-white dark:bg-stone-900">
   304→      <CardHeader className="pb-2 pt-4 px-4">
   305→        <CardTitle className="text-xs font-semibold text-stone-700 dark:text-stone-200 flex items-center gap-1.5">
   306→          <TrendingUp className="h-3.5 w-3.5 text-emerald-600" />
   307→          Audit Velocity
   308→        </CardTitle>
   309→        <CardDescription className="text-[10px]">Events per month (last 6 months)</CardDescription>
   310→      </CardHeader>
   311→      <CardContent className="px-4 pb-4">
   312→        <div className="h-28">
   313→          <ResponsiveContainer width="100%" height="100%">
   314→            <LineChart data={chartData}>
   315→              <CartesianGrid strokeDasharray="3 3" stroke="#e7e5e4" />
   316→              <XAxis
   317→                dataKey="month"
   318→                tick={{ fontSize: 10, fill: '#78716c' }}
   319→                axisLine={{ stroke: '#d6d3d1' }}
   320→                tickLine={false}
   321→              />
   322→              <YAxis
   323→                domain={[0, Math.max(maxEvents + 1, 4)]}
   324→                tick={{ fontSize: 10, fill: '#78716c' }}
   325→                axisLine={false}
   326→                tickLine={false}
   327→                width={24}
   328→              />
   329→              <RechartsTooltip
   330→                contentStyle={{
   331→                  backgroundColor: '#fff',
   332→                  border: '1px solid #d6d3d1',
   333→                  borderRadius: '6px',
   334→                  fontSize: '11px',
   335→                  padding: '4px 8px',
   336→                }}
   337→                formatter={(value: number) => [`${value} event${value !== 1 ? 's' : ''}`, 'Count']}
   338→              />
   339→              <Line
   340→                type="monotone"
   341→                dataKey="events"
   342→                stroke="#10b981"
   343→                strokeWidth={2}
   344→                dot={{ fill: '#10b981', r: 3 }}
   345→                activeDot={{ r: 5, stroke: '#10b981', strokeWidth: 2, fill: '#fff' }}
   346→              />
   347→            </LineChart>
   348→          </ResponsiveContainer>
   349→        </div>
   350→      </CardContent>
   351→    </Card>
   352→  );
   353→}

   392→function RiskForecastPanel({ project }: { project: ProjectRecord }) {
   393→  const forecast = useMemo((): RiskForecast => {
   394→    const riskScore = project.riskScore ?? 30;
   395→    const riskFactorCount = project.riskFactors?.length ?? 0;
   396→    const stallingProbability = Math.min(
   397→      95,
   398→      Math.max(5, Math.round((riskScore * 0.6) + (riskFactorCount * 8)))
   399→    );
   400→    const budgetVelocity = project.budgetAllocated > 0
   401→      ? Math.round((project.budgetSpent / project.budgetAllocated) * 100)
   402→      : 0;
   403→
   404→    return {
   405→      projectId: project.id,
   406→      stallingProbability,
   407→      factors: [
   408→        {
   409→          name: 'Budget Absorption',
   410→          weight: budgetVelocity > 80 ? 0.3 : budgetVelocity > 50 ? 0.15 : 0.35,
   411→          status: budgetVelocity > 80 ? 'normal' : budgetVelocity > 50 ? 'concerning' : 'critical',
   412→        },
   413→        {
   414→          name: 'Risk Score',
   415→          weight: riskScore > 60 ? 0.35 : riskScore > 30 ? 0.2 : 0.1,
   416→          status: riskScore > 60 ? 'critical' : riskScore > 30 ? 'concerning' : 'normal',
   417→        },
   418→        {
   419→          name: 'Risk Factor Count',
   420→          weight: Math.min(riskFactorCount * 0.1, 0.3),
   421→          status: riskFactorCount > 3 ? 'critical' : riskFactorCount > 1 ? 'concerning' : 'normal',
   422→        },
   423→      ],
   424→      predictedCompletionDate: project.endDate,
   425→      budgetVelocity,
   426→      milestoneCompletionRate: project.status === 'completed' ? 100 : project.status === 'active' ? 65 : 30,
   427→      recommendation:
   428→        stallingProbability > 70
   429→          ? 'Immediate intervention recommended. Escalate to OAG for priority review.'
   430→          : stallingProbability > 40
   431→          ? 'Monitor closely. Budget absorption below target — engage county assembly oversight.'
   432→          : 'Project on track. Continue routine monitoring and citizen auditing.',
   433→    };
   434→  }, [project]);
   435→
   436→  const stallingColor =
   437→    forecast.stallingProbability > 70
   438→      ? 'text-red-600'
   439→      : forecast.stallingProbability > 40
   440→      ? 'text-amber-600'
   441→      : 'text-emerald-600';
   442→
   443→  const stallingBarColor =
   444→    forecast.stallingProbability > 70
   445→      ? 'bg-red-500'
   446→      : forecast.stallingProbability > 40
   447→      ? 'bg-amber-500'
   448→      : 'bg-emerald-500';
   449→
   450→  return (
   451→    <Card className="border-stone-200 dark:border-stone-700 bg-white dark:bg-stone-900">
   452→      <CardHeader className="pb-2 pt-4 px-4">
   453→        <CardTitle className="text-xs font-semibold text-stone-700 dark:text-stone-200 flex items-center gap-1.5">
   454→          <Gauge className="h-3.5 w-3.5 text-amber-600" />
   455→          Risk Forecast
   456→        </CardTitle>
   457→        <CardDescription className="text-[10px]">Stalling probability analysis</CardDescription>
   458→      </CardHeader>
   459→      <CardContent className="px-4 pb-4 space-y-3">
   460→        {/* Stalling probability */}
   461→        <div className="flex items-center gap-2">
   462→          <div className="flex-1">
   463→            <div className="flex justify-between text-[11px] mb-1">
   464→              <span className="text-stone-600 dark:text-stone-300">Stalling Probability</span>
   465→              <span className={`font-bold ${stallingColor}`}>
   466→                {forecast.stallingProbability}%
   467→              </span>
   468→            </div>
   469→            <div className="h-1.5 rounded-full bg-stone-100 dark:bg-stone-700 overflow-hidden">
   470→              <div
   471→                className={`h-full rounded-full transition-all duration-700 ${stallingBarColor}`}
   472→                style={{ width: `${forecast.stallingProbability}%` }}
   473→              />
   474→            </div>
   475→          </div>
   476→        </div>
   477→
   478→        {/* Factors */}
   479→        <div className="space-y-1.5">
   480→          {forecast.factors.map((factor) => (
   481→            <div key={factor.name} className="flex items-center justify-between text-[11px]">
   482→              <span className="text-stone-500 dark:text-stone-400">{factor.name}</span>
   483→              <Badge
   484→                variant="outline"
   485→                className={`text-[9px] px-1.5 py-0 h-4 border ${
   486→                  factor.status === 'critical'
