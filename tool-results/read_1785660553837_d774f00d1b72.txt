    35→// ─── Avatar color helper ──────────────────────────────────────────

    37→function getAvatarColor(name: string): string {
    38→  const colors = [
    39→    'bg-blue-500', 'bg-emerald-500', 'bg-violet-500', 'bg-amber-500',
    40→    'bg-rose-500', 'bg-cyan-500', 'bg-indigo-500', 'bg-teal-500',
    41→    'bg-orange-500', 'bg-pink-500',
    42→  ];
    43→  let hash = 0;
    44→  for (let i = 0; i < name.length; i++) {
    45→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    46→  }
    47→  return colors[Math.abs(hash) % colors.length];
    48→}
    49→
    50→function getInitials(name: string): string {
    51→  return name
    52→    .split(' ')
    53→    .map(w => w[0])
    54→    .join('')
    55→    .toUpperCase()
    56→    .slice(0, 2);
    57→}
    58→
    59→function timeAgo(dateStr: string): string {
    60→  const date = new Date(dateStr);
    61→  const now = new Date();
    62→  const diffMs = now.getTime() - date.getTime();
    63→  const diffMins = Math.floor(diffMs / 60000);
    64→  const diffHours = Math.floor(diffMs / 3600000);
    65→  const diffDays = Math.floor(diffMs / 86400000);
    66→
    67→  if (diffMins < 1) return 'Just now';
    68→  if (diffMins < 60) return `${diffMins}m ago`;
    69→  if (diffHours < 24) return `${diffHours}h ago`;
    70→  if (diffDays < 7) return `${diffDays}d ago`;
    71→  return date.toLocaleDateString('en-KE', { day: 'numeric', month: 'short' });
    72→}

   484→function UploadForm({ projectId, onProofAdded, activeType }: {
   485→  projectId: string;
   486→  onProofAdded: () => void;
   487→  activeType: string;
   488→}) {
   489→  const [uploading, setUploading] = useState(false);
   490→  const [caption, setCaption] = useState('');
   491→  const [authorName, setAuthorName] = useState('');
   492→  const [authorLocation, setAuthorLocation] = useState('');
   493→  const [commentText, setCommentText] = useState('');
   494→  const [selectedFile, setSelectedFile] = useState<File | null>(null);
   495→  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
   496→  const fileInputRef = useRef<HTMLInputElement>(null);
   497→
   498→  const resetForm = () => {
   499→    setCaption('');
   500→    setCommentText('');
   501→    setSelectedFile(null);
   502→    setPreviewUrl(null);
   503→  };
   504→
   505→  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   506→    const file = e.target.files?.[0];
   507→    if (file) {
   508→      setSelectedFile(file);
   509→      if (file.type.startsWith('image/')) {
   510→        const url = URL.createObjectURL(file);
   511→        setPreviewUrl(url);
   512→      } else {
   513→        setPreviewUrl(null);
   514→      }
   515→    }
   516→  };
   517→
   518→  const handleUpload = async () => {
   519→    if (!authorName.trim()) {
   520→      toast.error('Please enter your name');
   521→      return;
   522→    }
   523→
   524→    if (activeType === 'comment') {
   525→      if (!commentText.trim()) {
   526→        toast.error('Please write a comment');
   527→        return;
   528→      }
   529→      // Submit text comment
   530→      try {
   531→        setUploading(true);
   532→        const res = await fetch(`/api/projects/${projectId}/proofs`, {
   533→          method: 'POST',
   534→          headers: { 'Content-Type': 'application/json' },
   535→          body: JSON.stringify({
   536→            type: 'comment',
   537→            content: commentText.trim(),
   538→            authorName: authorName.trim(),
   539→            authorLocation: authorLocation.trim(),
   540→          }),

   872→              <>
   873→                <Upload className="h-3 w-3" />
   874→                Add Proof
   875→              </>
   876→            )}
   877→          </Button>
   878→        </div>
